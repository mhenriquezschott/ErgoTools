import json,sys
from pathlib import Path
from ergotools.domain.assessments.tst import TstDirection,calculate_tst_task
from ergotools.domain.assessments.tst_draft import TstDirectionConvention,TstDraft,TstTask
from ergotools.domain.model_units import MeasurementSystem
from ergotools.infrastructure.tst_serialization import encode_tst,decode_tst
root=Path.cwd()
records=json.loads((root/'tests/fixtures/tst_calculation_reference.json').read_text())['records']
inverse={'0':TstDirection.HANDLING_LOADS,'1':TstDirection.HORIZONTAL,'2':TstDirection.DOWNWARD}
for row in records:
 source=TstDraft(MeasurementSystem(row['system']),(TstTask('x',inverse[row['direction']],row['lever_arm'],row['load'],row['repetitions']),),TstDirectionConvention.REFERENCE_UI)
 assert decode_tst(encode_tst(source))==source
 result=calculate_tst_task(source.model_task('x'),source.system)
 assert result.model_moment_ft_lbf==row['model_moment'] and result.displayed_moment==row['display_moment'] and result.cumulative_damage==row['damage']
sys.path.append(str(root))
exec(compile(open('/tmp/f137-view.py').read(),'/tmp/f137-view.py','exec'))
for name,module in tuple(sys.modules.items()):
 if name=='ergotools' or name.startswith('ergotools.'):
  filename=getattr(module,'__file__',None)
  if filename:assert Path(filename).is_relative_to(Path(sys.argv[1])),filename
print('Extracted-wheel F137 original dropdown-to-model mapping on all 1650 frozen vectors and source v1/v2/native controls passed.')

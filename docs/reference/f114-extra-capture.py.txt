import tempfile
from pathlib import Path

from PyQt5.QtTest import QTest
from ergotools.bootstrap import load_icons

from tests.test_plot_markers import canvas_with_map
from tests.test_plot_results import project,SCOPE
app=QApplication.instance() or QApplication([]);art=Path('artifacts/plot-graph-workflows')
with tempfile.TemporaryDirectory() as directory:
 gen=project.__wrapped__(Path(directory));repo=next(gen)
 try:
  canvas=canvas_with_map(Path(directory));layer=PlotMarkerLayer(canvas,repo.load_jobs(),repo.save_plot_positions,lambda msg:None,current_year=2026)
  for language,label in ((Language.ENGLISH,'en'),(Language.SPANISH,'es')):
   panel=PlotToolsOverview(layer,load_icons(),language=language);layer.set_results(saved_plot_results(repo,PlotFilterSelection(SCOPE),current_year=2026))
   panel.settings=PlotGraphSettings(show_grid=True,show_bar_values=True,axis_mode=PlotAxisMode.CUSTOM,axis_maximum=80,text_scale=140)
   panel.draw_current();panel.resize(330,600);panel.show();app.processEvents();QTest.qWait(100)
   assert panel.grab().save(str(art/(label+'-panel-values-140.png')))
   dialog=PlotGraphDialog(panel.settings,load_icons(),panel.save_plot,language=language,parent=panel);dialog.show();app.processEvents()
   QFileDialog.getSaveFileName=lambda *args,**kwargs:(str(Path(directory)/'missing'/'chart.png'),'PNG')
   def capture_message():
    modal=QApplication.activeModalWidget()
    assert modal is not None
    assert modal.grab().save(str(art/(label+'-settings-export-error.png')))
    modal.accept()
   QTimer.singleShot(100,capture_message);dialog.save_button.click();assert dialog.isVisible()
   dialog.close()
   viewer=PlotFigureViewer(panel.make_figure(),load_icons(),language=language,parent=panel);viewer.show();app.processEvents()
   def capture_viewer_error():
    modal=QApplication.activeModalWidget()
    assert modal is not None
    assert modal.grab().save(str(art/(label+'-viewer-export-error.png')))
    modal.accept()
   QTimer.singleShot(100,capture_viewer_error);viewer.save_button.click();assert viewer.isVisible();viewer.close()
   panel.close()
  canvas.close()
 finally:gen.close()
print('Localized largest-text narrow panel and both actual export error dialogs inspected')

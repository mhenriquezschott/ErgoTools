import ast
import hashlib
import json
import sys
from pathlib import Path
from types import SimpleNamespace

root = Path(sys.argv[1])
source = root / 'src/main.py'
names = {'setMetricSystem', 'setImperialSystem', 'updateUnits', 'setUnitText', 'setUnitSysText'}
nodes = [n for n in ast.walk(ast.parse(source.read_text())) if isinstance(n, ast.FunctionDef) and n.name in names]
assert len(nodes) == len(names)
namespace = {}
exec(compile(ast.Module(body=nodes, type_ignores=[]), str(source), 'exec'), namespace)
owner = type('ReferenceUnitMethods', (), {name: namespace[name] for name in names})
class Label:
    def __init__(self, value=''):
        self.value = value
    def setText(self, value):
        self.value = value
    def text(self):
        return self.value
records = []
for lifft in ('Metric', 'Imperial'):
    for tst in ('Metric', 'Imperial'):
        for tab in range(3):
            for target in ('Metric', 'Imperial'):
                obj = owner()
                obj.default_metric_sys = 'Metric'
                obj.lifft_unit, obj.tst_unit = lifft, tst
                obj.tabWidget = SimpleNamespace(currentIndex=lambda: tab)
                obj.unit_label = Label()
                for attr, count in (('lifft_headers_labels', 7), ('lifft_headers_labels_fixed', 7), ('tst_headers_labels', 8), ('tst_headers_labels_fixed', 8)):
                    setattr(obj, attr, [Label() for _ in range(count)])
                obj.lifft_lever_arm_inputs = [Label('12.345')]
                obj.lifft_load_inputs = [Label('10')]
                obj.tst_lever_arm_inputs = [Label('27')]
                obj.tst_load_inputs = [Label('22.25')]
                getattr(obj, 'set' + target + 'System')()
                records.append(dict(lifft_source=lifft, tst_source=tst, tab=tab, target=target,
                    selected=obj.selectedMeasurementSystem, lifft_recorded=obj.lifft_unit, tst_recorded=obj.tst_unit,
                    input_texts=[getattr(obj, attr)[0].text() for attr in ('lifft_lever_arm_inputs','lifft_load_inputs','tst_lever_arm_inputs','tst_load_inputs')],
                    lifft_headers=[label.text() for label in obj.lifft_headers_labels],
                    tst_headers=[label.text() for label in obj.tst_headers_labels],
                    unit_label=obj.unit_label.text()))
output = dict(reference_python=sys.version.split()[0], source_sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
              methods=sorted(names), method_ast_sha256=hashlib.sha256(ast.dump(ast.Module(body=nodes, type_ignores=[])).encode()).hexdigest(),
              note='Actual extracted methods with inert labels; no GUI or calculation invoked.', records=records)
Path(sys.argv[2]).write_text(json.dumps(output, indent=2)+'\n')
print(len(records), 'actual reference unit-menu cases captured')

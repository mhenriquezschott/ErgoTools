import hashlib,json,os,subprocess,sys,tempfile,zipfile
from pathlib import Path
root=Path.cwd()
def digest(p): return hashlib.sha256(p.read_bytes()).hexdigest()
ref=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
actual={str(p.relative_to(ref)):digest(p) for p in ref.rglob('*') if p.is_file() and '.git' not in p.relative_to(ref).parts}
assert actual==json.loads((root/'docs/reference/source-manifest.json').read_text())
assert digest(root/'uv.lock')==json.loads((root/'docs/f28-validation.json').read_text())['lock_sha256']
wheel=root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'
code='''
import sys
from pathlib import Path
sys.path.insert(0,sys.argv[1])
import ergotools
assert Path(ergotools.__file__).is_relative_to(Path(sys.argv[1]))
from ergotools.application.session import DocumentKey,DocumentSession
from ergotools.application.assessments.evaluation import AssessmentEvaluator
from ergotools.domain.assessments.lifft_draft import LifftDraft,LifftTask
from ergotools.domain.model_units import MeasurementSystem
from ergotools.presentation.lifft_results import present_lifft_results
for system in MeasurementSystem:
 session=DocumentSession(DocumentKey('p','lifft'),LifftDraft(system,(LifftTask('a',12,10,100),LifftTask('b',30,50,200))))
 outcome=AssessmentEvaluator().evaluate(session)
 display=present_lifft_results(outcome,session)
 if system is MeasurementSystem.METRIC:
  assert display is None
 else:
  assert [row.task_id for row in display.rows]==['a','b']
  assert display.rows[0].moment_text=='10.0'
  assert display.rows[1].moment_text=='125.0'
  assert present_lifft_results(outcome,DocumentSession(session.document,session.state)) is None
print('isolated result-presentation wheel smoke passed')
'''
with tempfile.TemporaryDirectory() as directory:
 with zipfile.ZipFile(wheel) as z: z.extractall(directory)
 r=subprocess.run([sys.executable,'-I','-c',code,directory],cwd=directory,check=True,capture_output=True,text=True);print(r.stdout.strip())
coverage=json.loads((root/'artifacts/coverage.json').read_text())
e=dict(milestone='F29',verified_on='2026-09-30',tests_passed=293,new_tests=5,reference_task_vectors=1075,reference_summary_groups=29,coverage_percent=coverage['totals']['percent_covered'],new_module_coverage_percent=100,reference_files_unchanged=len(actual),dependency_lock_unchanged=True,lock_sha256=digest(root/'uv.lock'),wheel_sha256=digest(wheel),independent_wheel_smoke='passed',ui_changed=False,overall_completion_estimate_percent=55)
(root/'docs/f29-validation.json').write_text(json.dumps(e,indent=2)+'\n');print(json.dumps(e,indent=2))

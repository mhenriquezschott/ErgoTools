import ast, hashlib, json, sqlite3, sys, tempfile
from contextlib import closing
from dataclasses import dataclass, astuple
from pathlib import Path
from types import SimpleNamespace
root=Path.cwd()
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
namespace={'sqlite3':sqlite3,'dataclass':dataclass}
sources={'src/main.py':['getWorkers'],'src/job_placement_repository.py':['WorkplaceKey','available_workplaces']}
for name, wanted in sources.items():
    tree=ast.parse((reference/name).read_text())
    nodes=[next(node for node in ast.walk(tree) if isinstance(node,(ast.FunctionDef,ast.ClassDef)) and node.name==item) for item in wanted]
    exec(compile(ast.Module(body=nodes,type_ignores=[]),name,'exec'),namespace)
workers=[['w 2','Ana','Zulu'],['001',None,None],['W1','Bea','alpha'],['w3','C','Zulu'],['empty','',''],['partial','Only',None],['accent','\u00c1na','\u00d1ez'],['spaces',' ',' ']]
stations=[['P','S','L','2'],['P','S','L','10'],['P','S2','L','2'],['p','S','L','2'],["P'",'S','L','2']]
shifts=['Night','Day']
cases=[]
with tempfile.TemporaryDirectory() as directory:
    path=Path(directory)/'fixture.sqlite'
    for name, station_rows, shift_rows in [('populated',stations,shifts),('no_shifts',stations,[]),('no_stations',[],shifts),('empty',[],[])]:
        with closing(sqlite3.connect(path)) as connection, connection:
            connection.executescript('DROP TABLE IF EXISTS Worker; DROP TABLE IF EXISTS Station; DROP TABLE IF EXISTS Shift; CREATE TABLE Worker(id TEXT PRIMARY KEY,first_name TEXT,last_name TEXT); CREATE TABLE Station(plant_name TEXT,section_name TEXT,line_name TEXT,id TEXT); CREATE TABLE Shift(id TEXT PRIMARY KEY);')
            connection.executemany('INSERT INTO Worker VALUES (?,?,?)',workers)
            connection.executemany('INSERT INTO Station VALUES (?,?,?,?)',station_rows)
            connection.executemany('INSERT INTO Shift VALUES (?)',[(s,) for s in shift_rows])
        owner=SimpleNamespace(projectFileCreated=True,projectdatabasePath=str(path))
        labels=[namespace['getWorkers'](owner,order) for order in (0,1)]
        with closing(sqlite3.connect(path)) as connection:
            contexts=[astuple(value) for value in namespace['available_workplaces'](connection)]
            ids=[[row[0] for row in connection.execute('SELECT id FROM Worker ORDER BY '+column)] for column in ('id','last_name')]
        cases.append(dict(name=name,workers=workers,stations=station_rows,shifts=shift_rows,expected_workplaces=contexts,expected_worker_labels=labels,expected_worker_ids=ids))
fixture=dict(python=sys.version.split()[0],source_sha256={name:hashlib.sha256((reference/name).read_bytes()).hexdigest() for name in sources},cases=cases,notes='Actual AST-extracted getWorkers and available_workplaces over temporary synthetic SQLite; label IDs captured separately without parsing. Equal-name order reflects this fixture, not a guaranteed SQL tie policy.')
(root/'tests/fixtures/work_catalog_reference.json').write_text(json.dumps(fixture,indent=2)+'\n')
print(len(cases),'catalog reference cases captured')

import hashlib
import json
import shutil
import subprocess
import sys
import tomllib
import zipfile
from pathlib import Path

r=Path.cwd();wheel=r/'dist/ergotools-0.1.0.dev0-py3-none-any.whl';dest=Path('/tmp/ergotools-m12-final-wheel');dest.mkdir(exist_ok=True)
with zipfile.ZipFile(wheel) as archive:
 archive.extractall(dest)
 paths=list((r/'src/ergotools').rglob('*.py'))
 for p in paths:assert archive.read('ergotools/'+str(p.relative_to(r/'src/ergotools')))==p.read_bytes()
 for lang in ('en','es'):assert archive.read(f'ergotools/resources/documents/user-guide-{lang}.html')==(r/f'src/ergotools/resources/documents/user-guide-{lang}.html').read_bytes()
code="""import sys;from pathlib import Path
sys.path.insert(0,sys.argv[1]);sys.path.append(str(Path.cwd()))
import pytest
result=pytest.main(['-q','-o','cache_dir=/tmp/ergotools-m12-wheel-pytest','tests/test_launcher.py','tests/test_help_workflow.py','tests/test_resources.py'])
assert result==0
assert all(not getattr(m,'__file__',None) or Path(m.__file__).is_relative_to(sys.argv[1]) for name,m in tuple(sys.modules.items()) if name=='ergotools' or name.startswith('ergotools.'))
print('All application imports confined to final wheel.')
"""
subprocess.run([sys.executable,'-I','-c',code,str(dest)],check=True,timeout=120)
expected={s.split('==')[0]:s.split('==')[1].split(';')[0].strip() for s in tomllib.loads((r/'pyproject.toml').read_text())['project']['optional-dependencies']['desktop']}
script='import importlib.metadata,json,sys; expected=json.loads(sys.argv[1]); actual={p:importlib.metadata.version(p) for p in expected}; assert actual==expected,(actual,expected); print(json.dumps(actual,indent=2))'
versions=subprocess.check_output(['/tmp/ergotools-m12-clean/env/bin/python','-I','-c',script,json.dumps(expected)],text=True)
(r/'artifacts/m12-delivery/installed-versions.json').write_text(versions)
ref=Path(json.loads((r/'docs/reference/provenance.json').read_text())['reference_root']);manifest=json.loads((r/'docs/reference/source-manifest.json').read_text())
assert all(hashlib.sha256((ref/p).read_bytes()).hexdigest()==digest for p,digest in manifest.items())
lock=hashlib.sha256((r/'uv.lock').read_bytes()).hexdigest();assert lock=='9e1f2f9f811ec3e9f57f435bd719d7e66112fb4977f6a29f0f845ce4dcaf39ae'
shutil.copyfile(wheel,r/'dist/linux'/wheel.name)
delivery=r/'dist/linux';(delivery/'SHA256SUMS').write_text(''.join(hashlib.sha256(p.read_bytes()).hexdigest()+'  '+p.name+'\n' for p in sorted(delivery.iterdir()) if p.is_file() and p.name!='SHA256SUMS'))
(r/'artifacts/m12-delivery/final-package.json').write_text(json.dumps({'wheel_sha256':hashlib.sha256(wheel.read_bytes()).hexdigest(),'source_python_files_equal':len(paths),'current_guides_equal':2,'reference_files_unchanged':len(manifest),'lock_sha256':lock,'exact_installed_runtime_packages':len(expected),'final_wheel_tests':21,'all_application_imports_confined_to_wheel':True,'final_guide_only_change_after_full_gate':True},indent=2)+'\n')
print('Final wheel, runtime versions, delivery checksums and reference/lock verified.')

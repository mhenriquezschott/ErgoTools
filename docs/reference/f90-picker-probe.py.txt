from pathlib import Path
import tempfile
from PyQt5.QtWidgets import QApplication
from PyQt5.QtCore import QTimer
from ergotools.bootstrap import create_window
app=QApplication([])
with tempfile.TemporaryDirectory() as directory:
 window=create_window(Path(directory)/'settings.json',original_navigation=True)
 window.show()
 for title in ('save_as','backup'):
  def reject():
   print('REJECT',title,flush=True)
   QApplication.activeModalWidget().reject()
  print('START',title,flush=True)
  QTimer.singleShot(250,reject)
  print('RETURN',window.choose_file(True,title_key=title),flush=True)
 window.close()

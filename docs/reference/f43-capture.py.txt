import runpy, sqlite3, json, hashlib
from contextlib import closing
from datetime import date
from pathlib import Path
root=Path.cwd()
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])/'src/job_placement_repository.py'
fn=runpy.run_path(str(reference))['update_worker_assignment_jobs']
class FixedDate(date):
 @classmethod
 def today(cls): return cls(2026,10,1)
fn.__globals__['date']=FixedDate
cases=[]
for label,initial in [('same',(1,2,'2020-01-01',None,'original')),('unclassified',(1,None,None,None,'original')),('classified',(1,1,'2020-01-01',None,'original')),('existing_bounds',(1,1,'2020-01-01','2026-09-01','original')),('remove',(1,1,'2020-01-01',None,'original')),('empty',(1,None,None,None,'original'))]:
 with closing(sqlite3.connect(':memory:')) as c:
  c.executescript("""CREATE TABLE Worker(id TEXT); INSERT INTO Worker VALUES('w');
CREATE TABLE Job(id TEXT, active INTEGER); INSERT INTO Job VALUES('j',1);
CREATE TABLE JobPlacement(id INTEGER,job_id TEXT,workplace_context_id INTEGER,active INTEGER); INSERT INTO JobPlacement VALUES(1,'j',1,1),(2,'j',1,1);
CREATE TABLE WorkerAssignment(id INTEGER PRIMARY KEY,worker_id TEXT,workplace_context_id INTEGER,job_placement_id INTEGER,started_at TEXT,ended_at TEXT,active INTEGER,notes TEXT);""")
  if initial:
   identity,placement,start,end,notes=initial
   c.execute('INSERT INTO WorkerAssignment VALUES(?,?,1,?,?,?,1,?)',(identity,'w',placement,start,end,notes))
  target=None if label in ('remove','empty') else 2
  result=fn(c,'w',{1:target})
  rows=c.execute('SELECT id,job_placement_id,started_at,ended_at,active,notes FROM WorkerAssignment ORDER BY id').fetchall()
  cases.append(dict(name=label,initial=initial,target=target,rows=rows))
(root/'tests/fixtures/job_classification_reference.json').write_text(json.dumps(dict(source_sha256=hashlib.sha256(reference.read_bytes()).hexdigest(),cases=cases),indent=2)+'\n')

from pathlib import Path
import tempfile
from unittest.mock import patch
from PyQt5.QtWidgets import QApplication,QMessageBox
from PyQt5.QtTest import QTest
from PyQt5.QtCore import QPoint,QTimer
from PyQt5.QtGui import QPainter,QPixmap
from ergotools.presentation.desktop.task_count_dialog import TaskCountDialog
from ergotools.bootstrap import create_window
from ergotools.application.document_lifecycle import CloseChoice
from ergotools.application.settings import Language
from ergotools.domain.assessments.duet_draft import DuetDraft,DuetTask
from ergotools.domain.assessments.tst import TstTask,TstDirection
from ergotools.domain.assessments.tst_draft import TstDraft
from ergotools.domain.assessments.lifft_draft import LifftDraft,LifftTask
from ergotools.domain.model_units import MeasurementSystem
from ergotools.domain.project import ProjectMetadata
app=QApplication([]);out=Path.cwd()/'artifacts/task-count-integration';out.mkdir(parents=True,exist_ok=True)
def capture(window,name):
 QTest.qWait(900);engine=window._body_renderer
 from vtkmodules.vtkRenderingCore import vtkWindowToImageFilter
 from vtkmodules.vtkIOImage import vtkPNGWriter
 pixmap=window.grab();engine.render_window.Render()
 filt=vtkWindowToImageFilter();filt.SetInput(engine.render_window);filt.ReadFrontBufferOff();filt.Update()
 writer=vtkPNGWriter();buffer=out/(name+'-vtk.png');writer.SetFileName(str(buffer));writer.SetInputConnection(filt.GetOutputPort());writer.Write()
 painter=QPainter(pixmap);painter.drawPixmap(engine.widget.mapTo(window,QPoint(0,0)),QPixmap(str(buffer)));painter.end();pixmap.save(str(out/(name+'.png')))
with tempfile.TemporaryDirectory() as directory:
 window=create_window(Path(directory)/'settings.json',original_navigation=True,enable_vtk=True);workspace=window.presenter.workspace
 workspace.create(str(Path(directory)/'tools.ergprj'),ProjectMetadata('p','Integrated tools',MeasurementSystem.IMPERIAL),'lifft',LifftDraft(MeasurementSystem.IMPERIAL,(LifftTask('l',12,20,100),)))
 workspace.create_assessment('duet',DuetDraft(tuple(DuetTask(str(i),6,100) for i in range(15))))
 workspace.create_assessment('st',TstDraft(MeasurementSystem.IMPERIAL,tuple(TstTask(str(i),TstDirection.DOWNWARD,12,20,100) for i in range(15))))
 window.refresh();window.show()
 for index,kind in ((1,'duet'),(2,'st')):
  with patch.object(window,'dirty_choice',return_value=CloseChoice.DISCARD):window.assessment_tabs.setCurrentIndex(index)
  window.language=Language.ENGLISH if index==1 else Language.SPANISH;window.translate();window.refresh()
  panel=window.secondary_panel();field=panel.inputs[0][-1];field.setText('bad');field.textEdited.emit('bad')
  def cancel():
   modal=QApplication.activeModalWidget();assert isinstance(modal,TaskCountDialog)
   QTest.qWait(100);modal.grab().save(str(out/(kind+'-modal-default.png')))
   modal.count.setValue(100);modal.count.selectAll();QTest.qWait(100);modal.grab().save(str(out/(kind+'-modal-selected-100.png')))
   modal.cancel_button.click()
  QTimer.singleShot(0,cancel);window.main_navigation.toolbar_actions['settings'].trigger();capture(window,kind+'-cancel-pending')
  def accept_one():
   modal=QApplication.activeModalWidget();modal.count.setValue(1);modal.set_button.click()
  with patch.object(QMessageBox,'information'):
   QTimer.singleShot(0,accept_one);window.main_navigation.toolbar_actions['settings'].trigger()
  capture(window,kind+'-one-row')
  window.run(window.undo);capture(window,kind+'-undo-populated')
  def accept_max():
   modal=QApplication.activeModalWidget();modal.count.setValue(100);modal.set_button.click()
  with patch.object(QMessageBox,'information'):
   QTimer.singleShot(0,accept_max);window.main_navigation.toolbar_actions['settings'].trigger()
  QTest.qWait(150);panel.body_scroll.verticalScrollBar().setValue(panel.body_scroll.verticalScrollBar().maximum());capture(window,kind+'-100-bottom')
  window.navigation_action(window.save_project)
 workspace.close(choice=CloseChoice.DISCARD);window.close();QTest.qWait(100)
 print('Native DUET/ST task-count modal, selected limits, Cancel/pending, accepted blank grids, Undo and 100-row bottom states captured')

import hashlib,json,shutil,tempfile
from collections import Counter
from pathlib import Path
from ergotools.infrastructure.legacy_project import read_legacy_descriptor,LegacyProject
from ergotools.infrastructure.legacy_catalogs import read_legacy_catalogs
from ergotools.infrastructure.legacy_assessments import read_legacy_assessments,LegacyShoulderSource
from ergotools.domain.assessments.document import AssessmentKind
root=Path.cwd();reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
source=reference/'tests/ErgoTools_ComparativeRiskTest.ergprj';original=read_legacy_descriptor(source,'comprehensive-validation')
paths=[source,original.database_path];before={str(p.relative_to(reference)):hashlib.sha256(p.read_bytes()).hexdigest() for p in paths}
with tempfile.TemporaryDirectory(prefix='ergotools-f130-') as directory:
 target=Path(directory)/source.name;shutil.copyfile(source,target)
 descriptor=read_legacy_descriptor(target,'comprehensive-validation');descriptor.database_path.parent.mkdir(parents=True);shutil.copyfile(original.database_path,descriptor.database_path)
 copied_hash=hashlib.sha256(descriptor.database_path.read_bytes()).hexdigest()
 with LegacyProject(descriptor) as project:
  catalogs=read_legacy_catalogs(project);assessments=read_legacy_assessments(project,catalogs)
  originals={r['id']:r for r in project.rows('IndividualAssessment')}
  for assessment in assessments:
   row=originals[assessment.source_id]
   assert assessment.ownership.assignment_id==row['worker_assignment_id']
   assert assessment.ownership.version==row['version']
   assert assessment.ownership.is_current==bool(row['is_current'])
   assert assessment.ownership.status.value==row['status']
   assert assessment.unit==row['unit']
   assert assessment.cumulative_damage==row['total_cumulative_damage']
   assert assessment.probability_percent==row['probability_outcome']
   assert assessment.notes==row['notes'] and assessment.assessed_at==row['assessed_at']
   for task,evidence,raw in zip(assessment.source.tasks,assessment.task_evidence,assessment.original_tasks,strict=True):
    assert task.task_id=='legacy-task-'+str(raw['task_index'])
    assert evidence.task_index==raw['task_index']
    assert evidence.cumulative_damage==raw['cumulative_damage'] and evidence.percentage_total==raw['percentage_total']
    assert evidence.moment==(None if assessment.ownership.kind is AssessmentKind.DUET else raw['moment'])
    assert task.repetitions==(None if raw['repetitions']=='' else raw['repetitions'])
    if isinstance(assessment.source,LegacyShoulderSource):assert task.selector_index==raw['type_of_task']
  tools=Counter(a.ownership.kind.value for a in assessments)
  task_counts=Counter()
  for assessment in assessments:task_counts[assessment.ownership.kind.value]+=len(assessment.source.tasks)
  raw_blank_inputs=sum(raw[name]=='' for assessment in assessments for raw in assessment.original_tasks for name in ('lever_arm','load','repetitions') if name in raw)
  # Compare old pre-normalization rows with normalized versioned ownership. These
  # remain separate evidence, never silently copied into the normalized history.
  contexts={r['id']:tuple(r[k] for k in ('plant_name','section_name','line_name','station_id','shift_id')) for r in project.rows('WorkplaceContext')}
  assignments={r['id']:(r['worker_id'],*contexts[r['workplace_context_id']]) for r in project.rows('WorkerAssignment')}
  normalized={( *assignments[r['worker_assignment_id']],r['tool_id']) for r in originals.values()}
  legacy_rows=project.rows('WorkerStationShiftErgoTool')
  fields=('worker_id','plant_name','section_name','line_name','station_id','shift_id','tool_id')
  unmatched=sum(tuple(r[k] for k in fields) not in normalized for r in legacy_rows)
  legacy_task_count=sum(len(project.rows(table)) for table in ('LifftResults','DuetResults','TstResults'))
 assert hashlib.sha256(descriptor.database_path.read_bytes()).hexdigest()==copied_hash
for path in paths:assert hashlib.sha256(path.read_bytes()).hexdigest()==before[str(path.relative_to(reference))]
record={'checkpoint':'F130','source_hashes':before,'normalized_assessments':len(assessments),'assessments_by_tool':dict(tools),'tasks_by_tool':dict(task_counts),'tasks_total':sum(task_counts.values()),'exact_cached_result_and_ownership_comparison':'Every normalized assessment and task compared against source values; all equal.','empty_string_input_fields_retained_in_raw_rows':raw_blank_inputs,'old_unversioned_summary_rows':len(legacy_rows),'old_unversioned_context_tool_rows_without_normalized_history':unmatched,'old_unversioned_task_rows':legacy_task_count,'limits':['ST selector indices retained raw; no TstDirection inference or scientific approval.','No cached results made fresh and no destination project published.','Old unversioned tables coexist with normalized history; their residual records must be explicitly reconciled before complete conversion.']}
(root/'docs/f130-legacy-audit.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record,indent=2))

"""Native history acceptance on a disposable comprehensive project copy."""
import hashlib
import json
import shutil
import traceback
from pathlib import Path
from unittest.mock import patch

from PyQt5.QtCore import QPoint, Qt, QTimer
from PyQt5.QtGui import QImage, QPainter
from PyQt5.QtTest import QTest
from PyQt5.QtWidgets import QApplication, QMessageBox
from vtkmodules.vtkIOImage import vtkPNGWriter
from vtkmodules.vtkRenderingCore import vtkWindowToImageFilter

from ergotools.bootstrap import create_window
from ergotools.domain.assessments.document import AssessmentKind
from ergotools.domain.plot_filters import PlotRiskView
from ergotools.presentation.desktop.plot_window import PlotWindow
from ergotools.presentation.desktop.rotation_dialog import RotationDialog
from ergotools.presentation.desktop.rotation_scope_dialog import RotationScopeDialog

root = Path.cwd()
out = root / 'artifacts/analysis-history-native'
out.mkdir(exist_ok=True)
stage = Path('/tmp/ergotools-analysis-history')
stage.mkdir(exist_ok=True)
project = stage / 'History validation copy.ergprj'
shutil.copyfile(root / 'artifacts/m10-integration-final/run/M10-validated.ergprj', project)
before_hash = hashlib.sha256(project.read_bytes()).hexdigest()
settings = stage / 'settings.json'
settings.unlink(missing_ok=True)
report = {'checks': {}, 'captures': [], 'failures': []}
app = QApplication([])
window = create_window(settings, original_controls=True, original_navigation=True, enable_vtk=True)

def capture(widget, name, *, vtk=False):
    app.processEvents()
    QTest.qWait(100)
    if isinstance(widget, PlotWindow):
        for gauge in (widget.outcome.gauge, widget.outcome.individual_gauge, widget.outcome.job_gauge):
            if gauge.has_value:
                gauge.animation.setCurrentTime(gauge.animation.duration())
            gauge.animation.stop()
    if vtk:
        for component in (window.results_sidebar.risk_gauge, window.results_sidebar.damage_progress):
            if component.has_value:
                component.animation.setCurrentTime(component.animation.duration())
            component.animation.stop()
        window._body_renderer._stop_animation()
        image = QImage(widget.size(), QImage.Format_ARGB32_Premultiplied)
        image.fill(Qt.transparent)
        widget.render(image, QPoint())
        renderer = window._body_renderer
        renderer.render_window.Render()
        filt = vtkWindowToImageFilter()
        filt.SetInput(renderer.render_window)
        filt.ReadFrontBufferOff()
        filt.Update()
        writer = vtkPNGWriter()
        writer.SetInputConnection(filt.GetOutputPort())
        writer.SetFileName(str(out / 'vtk-frame.png'))
        writer.Write()
        viewport = window.body_panel.viewport
        rect = viewport.geometry()
        rect.moveTopLeft(viewport.mapTo(window, QPoint()))
        painter = QPainter(image)
        painter.drawImage(rect, QImage(str(out / 'vtk-frame.png')))
        painter.end()
    else:
        image = widget.grab()
    assert image.save(str(out / f'{name}.png'))
    report['captures'].append(name)

def shortcut(view, redo=False):
    view.activateWindow()
    view.raise_()
    QTest.qWait(150)
    if isinstance(view, PlotWindow):
        view.map_panel.canvas.setFocus()
    elif isinstance(view, RotationDialog):
        view.grid.setFocus()
    else:
        view.assessment_tabs.setFocus()
    app.processEvents()
    QTest.keyClick(view, Qt.Key_Z, Qt.ControlModifier | (Qt.ShiftModifier if redo else Qt.NoModifier))
    app.processEvents()

def menu_capture(view, menu, name):
    menu.popup(view.mapToGlobal(QPoint(55, 25)))
    capture(menu, name)
    menu.hide()

def plot_exercise(language):
    view = app.activeModalWidget()
    try:
        assert isinstance(view, PlotWindow)
        view.resize(1460, 1060)
        capture(view, f'plot-{language}-initial')
        history = view.layer.shared_history
        initial = history.session.state
        filters = view.filters
        filters.demographics.setChecked(True)
        filters.ranges['age'][0].setValue(18)
        filters.ranges['age'][1].setValue(70)
        filters.ranges['height'][0].setValue(65.2)
        filters.apply_button.click()
        enabled = history.session.state
        view.resize(1800, 1200)
        capture(view, f'plot-{language}-demographics-enabled')
        filters.demographics.setChecked(False)
        filters.apply_button.click()
        disabled = history.session.state
        filters.tools.buttons[AssessmentKind.DUET].click()
        selected = history.session.state
        filters.apply_button.click()
        applied = history.session.state
        filters.risk_buttons[PlotRiskView.JOB].click()
        capture(view, f'plot-{language}-job-disabled-demographics')
        for expected in (applied, selected, disabled, enabled, initial):
            shortcut(view)
            assert history.session.state == expected
        shortcut(view, redo=True)
        assert filters.demographics.isChecked()
        assert filters.snapshot() == enabled.filters
        menu_capture(view, view.history_actions.menu_bar.actions()[0].menu(), f'plot-{language}-edit-menu')
        report['checks'][f'plot-{language}'] = True
    except Exception:
        report['failures'].append(traceback.format_exc())
    finally:
        view.reject()

def jrot_exercise(language):
    view = app.activeModalWidget()
    try:
        assert isinstance(view, RotationDialog)
        view.resize(1480, 740)
        initial = view.history.state
        tool = AssessmentKind.DUET if view.kind is not AssessmentKind.DUET else AssessmentKind.LIFFT
        view.tool_buttons[tool].click()
        selected = view.history.state
        def accept_scope():
            scope = app.activeModalWidget()
            assert isinstance(scope, RotationScopeDialog)
            first = next(iter(scope._contexts))
            scope._items[first].setCheckState(0, Qt.Checked)
            scope.use_button.click()
        QTimer.singleShot(100, accept_scope)
        view.choose_scope_button.setEnabled(True)
        view.choose_scope_button.click()
        pending = view.history.state
        assert pending.pending_scope
        capture(view, f'jrot-{language}-pending-scope')
        with patch.object(view, '_message'):
            view.apply_scope_button.click()
        assert view.history.state.scope == pending.pending_scope
        shortcut(view)
        assert view.history.state == pending
        shortcut(view)
        assert view.history.state == selected
        shortcut(view)
        assert view.history.state == initial
        shortcut(view, redo=True)
        assert view.history.state == selected
        view.resize(1800, 1000)
        capture(view, f'jrot-{language}-tool-redone')
        menu_capture(view, view.history_actions.menu_bar.actions()[0].menu(), f'jrot-{language}-edit-menu')
        report['checks'][f'jrot-{language}'] = True
    except Exception:
        report['failures'].append(traceback.format_exc())
    finally:
        view.reject()

def exercise():
    try:
        label = next(label for identity, label in window.document_options(str(project), window.language) if identity == 'legacy-assessment-2')
        with patch.object(window, 'choose_item', return_value=(label, True)):
            window.open_project_location(str(project))
        active = window.presenter.workspace.active
        source = active.state
        app.processEvents()
        window.task_panel.reset_button.click()
        assert active.state != source, "Reset did not change native source"
        shortcut(window)
        assert active.state == source
        shortcut(window, redo=True)
        assert active.state != source
        shortcut(window)
        assert active.state == source
        report['checks']['main-keyboard'] = True
        for index, language in enumerate(('en', 'es')):
            window.languages.setCurrentIndex(index)
            app.processEvents()
            capture(window, f'main-{language}', vtk=True)
            menu_capture(window, window._assessment_menu.edit, f'main-{language}-edit-menu')
            QTimer.singleShot(150, lambda language=language: plot_exercise(language))
            window.run(window.open_plot)
            QTimer.singleShot(150, lambda language=language: jrot_exercise(language))
            window.run(window.open_rotation)
        assert hashlib.sha256(project.read_bytes()).hexdigest() == before_hash
        report['checks']['project-bytes-unchanged'] = True
    except Exception:
        report['failures'].append(traceback.format_exc())
    finally:
        (out / 'verification.json').write_text(json.dumps(report, indent=2))
        app.quit()

window.show()
QTimer.singleShot(500, exercise)
QTimer.singleShot(90000, app.quit)
app.exec()
assert not report['failures'] and len(report['checks']) == 6, report
print(json.dumps(report))

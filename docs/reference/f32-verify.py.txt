import hashlib, json, subprocess, sys, tempfile, zipfile
from pathlib import Path
root=Path.cwd()
def digest(path): return hashlib.sha256(path.read_bytes()).hexdigest()
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
actual={str(p.relative_to(reference)):digest(p) for p in reference.rglob('*') if p.is_file() and '.git' not in p.relative_to(reference).parts}
assert actual==json.loads((root/'docs/reference/source-manifest.json').read_text())
assert digest(root/'uv.lock')==json.loads((root/'docs/f31-validation.json').read_text())['lock_sha256']
wheel=root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'
code='''
import sys
from pathlib import Path
sys.path.insert(0,sys.argv[1])
import ergotools
assert Path(ergotools.__file__).is_relative_to(Path(sys.argv[1]))
from ergotools.application.assessment_binding import AssessmentBinding
from ergotools.application.session import DocumentKey
from ergotools.domain.assessments.context import AssessmentContext, WorkerContext, WorkplaceContext
from ergotools.domain.assessments.document import AssessmentKind
from ergotools.domain.assessments.lifft_draft import LifftDraft
from ergotools.domain.model_units import MeasurementSystem
from ergotools.domain.project import ProjectMetadata
from ergotools.infrastructure.assessment_project_file import AssessmentProjectFile
path=Path('smoke.sqlite')
metadata=ProjectMetadata('p','Smoke',MeasurementSystem.IMPERIAL)
key=DocumentKey('p','a')
context=AssessmentContext(WorkerContext('p','w',WorkplaceContext('p','s','l','s','day')),AssessmentKind.LIFFT)
binding=AssessmentBinding(key,context)
source=LifftDraft(MeasurementSystem.IMPERIAL)
with AssessmentProjectFile.create_contextual(path,metadata) as project:
 project.insert_assessment(key,source)
 project.bind_assessment(binding)
 project.bind_assessment(binding)
with AssessmentProjectFile.open(path) as project:
 assert project.format_version==3
 assert project.assessment_binding(key)==binding
 assert project.find_assessment(context)==key
 assert project.load(key)==source
from dataclasses import replace
from ergotools.application.project_workspace import ProjectWorkspace
from ergotools.application.document_lifecycle import CloseChoice
from ergotools.application.assessments.lifft_commands import AddLifftTask
from ergotools.application.assessments.project_edits import LifftProjectEdit
from ergotools.domain.assessments.lifft_draft import LifftTask
from ergotools.infrastructure.settings_store import JsonSettingsStore
other=replace(context,worker=replace(context.worker,worker_id='other'))
second=DocumentKey('p','b')
with AssessmentProjectFile.open(path) as project:
 project.insert_assessment(second,source)
 project.bind_assessment(AssessmentBinding(second,other))
workspace=ProjectWorkspace(lambda location:AssessmentProjectFile.open(Path(location)),JsonSettingsStore(Path('settings.json')))
try:
 workspace.switch(str(path),key.document_id)
 old=workspace.active
 old.execute(LifftProjectEdit(AddLifftTask(key,LifftTask('task',12,10,100),0)))
 assert not workspace.select_context(other).completed
 assert workspace.active is old and old.is_dirty
 assert workspace.select_context(other,choice=CloseChoice.SAVE).completed
 assert not old.is_dirty and workspace.active.document==second
 assert workspace.select_context(context).completed
 assert workspace.active.state==old.state
finally:
 workspace.close(choice=CloseChoice.DISCARD)
print('isolated workspace context selection wheel smoke passed')
'''
with tempfile.TemporaryDirectory() as directory:
 with zipfile.ZipFile(wheel) as archive: archive.extractall(directory)
 result=subprocess.run([sys.executable,'-I','-c',code,directory],cwd=directory,check=True,capture_output=True,text=True)
 print(result.stdout.strip())
coverage=json.loads((root/'artifacts/coverage.json').read_text())
modules={name:data['summary']['percent_covered'] for name,data in coverage['files'].items() if name.endswith(('/project_workspace.py',))}
assert len(modules)==1 and all(value==100 for value in modules.values())
evidence=dict(milestone='F32',verified_on='2026-09-30',python=sys.version.split()[0],tests_passed=325,new_tests=10,coverage_percent=coverage['totals']['percent_covered'],workspace_module_coverage=modules,reference_files_unchanged=len(actual),dependency_lock_unchanged=True,lock_sha256=digest(root/'uv.lock'),wheel_sha256=digest(wheel),independent_wheel_smoke='passed',ui_changed=False,schema_changed=False,existing_formats_unchanged=True,overall_completion_estimate_percent=57)
(root/'docs/f32-validation.json').write_text(json.dumps(evidence,indent=2)+'\n')
print(json.dumps(evidence,indent=2))

import ast
import json
import os
import sqlite3
from datetime import date
from pathlib import Path
from types import SimpleNamespace

from PyQt5 import QtCore, QtGui, QtWidgets
from PyQt5.QtCore import QDate, QSize, Qt
from PyQt5.QtGui import QIcon
from PyQt5.QtWidgets import QApplication, QMainWindow
from ergotools.application.history_access import AssignmentSelection
from ergotools.domain.assessments.document import AssessmentKind
from ergotools.domain.language import Language
from ergotools.presentation.desktop.worker_search_dialog import WorkerSearchDialog, SearchAssessment
from ergotools.infrastructure.resources import read_asset
from tests.test_history_persistence import HIERARCHY
from tools.capture_desktop import render_widget

root=Path.cwd();out=root/'artifacts/m11-search';out.mkdir(exist_ok=True)
ref=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
source=ast.parse((ref/'src/main.py').read_text())
original=next(n for n in source.body if isinstance(n,ast.ClassDef) and n.name=='WorkerSearchDialog')
main=next(n for n in source.body if isinstance(n,ast.ClassDef) and n.name=='ErgoTools')
style_method=next(n for n in main.body if isinstance(n,ast.FunctionDef) and n.name=='mainWorkspaceStyleSheet')
original_style=next(n.value for n in ast.walk(style_method) if isinstance(n,ast.Constant) and isinstance(n.value,str) and 'QMainWindow' in n.value)
ns={**vars(QtWidgets),**vars(QtCore),**vars(QtGui),'os':os,'sqlite3':sqlite3,'__file__':str(ref/'src/main.py')}
exec(compile(ast.Module(body=[original],type_ignores=[]),'original-search-read-only','exec'),ns)
Original=ns['WorkerSearchDialog']
app=QApplication.instance() or QApplication([])
icons={}
for name in ('search','cancel','worker','plant','section','line','station'):
 pixmap=QtGui.QPixmap();pixmap.loadFromData(read_asset(f'icons/{name}.png'));icons[name]=QIcon(pixmap)
rows=tuple(SearchAssessment(AssignmentSelection('p',i,AssessmentKind.LIFFT),wid,last,first,sex,age,('P','S','L','1','Day'))
           for i,wid,last,first,sex,age in ((1,'w','Adams','Ana','Female','36'),(3,'x','Baker','Ben','Male','41')))
def original_load(self):
 self.worker_rows=[(r.worker_id,r.last_name,r.first_name,r.sex,r.age) for r in rows]
 self.worker_contexts={r.worker_id:{r.context} for r in rows}
 with sqlite3.connect(':memory:') as conn:
  conn.executescript("CREATE TABLE Plant(name); INSERT INTO Plant VALUES('P'); CREATE TABLE Section(name,plant_name); INSERT INTO Section VALUES('S','P'); CREATE TABLE Line(name,plant_name,section_name); INSERT INTO Line VALUES('L','P','S'); CREATE TABLE Station(id,plant_name,section_name,line_name); INSERT INTO Station VALUES('1','P','S','L');")
  self.populateWorkplaceTree(conn)
 for letter,button in self.alphabet_buttons.items():button.setEnabled(letter=='All' or letter in ('A','B'))
 self.applyFilters()
Original.loadData=original_load
host=QMainWindow();host.tabWidget=SimpleNamespace(currentIndex=lambda:0);host.mainWorkspaceStyleSheet=lambda:original_style
host.setStyleSheet(original_style)
old=Original(host)
new=WorkerSearchDialog(rows,HIERARCHY,AssessmentKind.LIFFT,icons,parent=host)
def capture(dialog,name):
 dialog.show();app.processEvents();render_widget(dialog).save(str(out/(name+'.png')))
def rect(widget):return [widget.x(),widget.y(),widget.width(),widget.height()]
comparisons=[]
for name,size in (('opening',(1080,650)),('minimum',(900,560)),('enlarged',(1500,850))):
 old.resize(*size);new.resize(*size)
 capture(old,'original-'+name);capture(new,'new-en-'+name)
 fields=('workplace_tree','table','search_input','use_button','result_count')
 comparison={field:{'original':rect(getattr(old,field)), 'new':rect(getattr(new,field))} for field in fields}
 comparisons.append({'size':name,'requested':size,'actual_original':[old.width(),old.height()], 'actual_new':[new.width(),new.height()],'controls':comparison})
new.resize(1080,650);new.workplace_tree.expandAll();new.table.selectRow(0);capture(new,'new-en-selected-expanded')
new.search_input.setText('no matching worker');capture(new,'new-en-empty')
spanish=WorkerSearchDialog(rows,HIERARCHY,AssessmentKind.LIFFT,icons,language=Language.SPANISH,parent=host)
for name,size in (('opening',(1080,650)),('minimum',(900,560)),('enlarged',(1500,850))):
 spanish.resize(*size);spanish.workplace_tree.expandAll();spanish.table.selectRow(0);capture(spanish,'new-es-'+name)
spanish.search_input.setText('no matching worker');capture(spanish,'new-es-empty')
(out/'comparison.json').write_text(json.dumps({'baseline':'Linux XCB approved Python/PyQt; original constructor/build/filter methods compiled from read-only AST; only SQL loader replaced by equivalent synthetic rows. No original module imports or application writes.','comparisons':comparisons,'captures':sorted(p.name for p in out.glob('*.png'))},indent=2)+'\n')
for dialog in (old,new,spanish):dialog.close()
host.close();app.processEvents()
print(json.dumps(comparisons))

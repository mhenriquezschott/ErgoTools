stage=Path('/tmp/ergotools-m12-clean');out=stage/'evidence';out.mkdir(exist_ok=True)
source=stage/'restored-original-copy.ergprj';backup=stage/'offline-backup.ergprj'
assert sys.version_info[:3]==(3,13,14)

def digest(path):
 with sqlite3.connect(f'file:{path}?mode=ro',uri=True) as conn:
  return hashlib.sha256('\n'.join(conn.iterdump()).encode()).hexdigest()
before=digest(source)
(stage/'preferences.json').unlink(missing_ok=True)
if backup.exists(): backup.unlink()
unrelated=stage/'unrelated launch folder';unrelated.mkdir(exist_ok=True);os.chdir(unrelated)
base_create=bootstrap.create_window
failures=[]
def capture(window,name):
 QApplication.processEvents();QTest.qWait(150)
 for component in (window.results_sidebar.risk_gauge,window.results_sidebar.damage_progress):
  if component.has_value: component.animation.setCurrentTime(component.animation.duration())
  component.animation.stop()
 window._body_renderer._stop_animation()
 image=QImage(window.size(),QImage.Format_ARGB32_Premultiplied);image.fill(Qt.transparent);window.render(image,QPoint())
 renderer=window._body_renderer
 assert renderer is not None
 renderer.render_window.Render()
 filt=vtkWindowToImageFilter();filt.SetInput(renderer.render_window);filt.ReadFrontBufferOff();filt.Update()
 writer=vtkPNGWriter();writer.SetInputConnection(filt.GetOutputPort());writer.SetFileName(str(out/'native-frame.png'));writer.Write()
 frame=QImage(str(out/'native-frame.png'));assert not frame.isNull()
 widget=window.body_panel.viewport
 rect=widget.geometry();rect.moveTopLeft(widget.mapTo(window,QPoint()))
 painter=QPainter(image);painter.drawImage(rect,frame);painter.end()
 assert image.save(str(out/(name+'.png')))
 report['captures'].append(name+'.png')
def exercise(window):
 try:
  assert QApplication.platformName()=='xcb'
  window.resize(1550,1015);capture(window,'01-installed-empty')
  assert window.main_navigation is not None and window.task_panel is not None
  assert window._body_renderer is not None
  with AssessmentProjectFile.open(source) as project:
   entry=next(e for e in project.list_assessments() if e.kind is AssessmentKind.LIFFT)
   expected=(project.load_hierarchy(),project.load_jobs(),project.load_history(),project.load_plot_positions(),project.load_rotation_schemes(),project.load_legacy_evidence())
  workspace=window.presenter.workspace
  workspace.switch(str(source),entry.document.document_id);window.refresh()
  capture(window,'02-installed-populated')
  window.languages.setCurrentIndex(list(Language).index(Language.SPANISH));capture(window,'03-installed-spanish')
  with patch.object(window,'choose_file',return_value=str(backup)):
   window.navigation_action(window.backup_project)
  assert backup.exists() and workspace.location==str(source)
  assert digest(backup)==before and digest(source)==before
  workspace.close(choice=CloseChoice.DISCARD);window.refresh()
  options=window.document_options(str(backup),window.language)
  label=next(label for identity,label in options if identity==entry.document.document_id)
  with patch.object(window,'choose_item',return_value=(label,True)):
   window.open_project_location(str(backup))
  capture(window,'04-installed-backup-reopened')
  with AssessmentProjectFile.open(backup) as project:
   actual=(project.load_hierarchy(),project.load_jobs(),project.load_history(),project.load_plot_positions(),project.load_rotation_schemes(),project.load_legacy_evidence())
   assert actual==expected
  values=((10,90),(80,20),(20,60))
  problem=RotationProblem((WorkerRotation('one',('a','a')),WorkerRotation('two',('b','b'))),tuple(RotationRisk(key,kind,value) for kind,risks in zip(AssessmentKind,values,strict=True) for key,value in zip(('a','b'),risks,strict=True)),5)
  optimizer=PulpRotationOptimizer();single=optimizer.optimize_single_tool(problem,AssessmentKind.LIFFT);multi=optimizer.optimize_all_tools(problem)
  problem.validate_schedule(single.schedule);problem.validate_schedule(multi.schedule)
  assert single.backend.value=='cbc' and multi.backend.value=='highs'
  report['checks']={'integrated_launcher':True,'native_vtk':True,'copied_original_project':True,'backup_exact_database_dump_sha256':before,'backup_reopened_all_owned_data_equal':True,'cbc':str(single.stages),'highs':str(multi.stages),'no_source_checkout_imports':True,'offline_network_namespace':True}
  assert read_asset('documents/user-guide-en.html') and read_asset('documents/user-guide-es.html')
  workspace.close(choice=CloseChoice.DISCARD);window.close()
 except BaseException:
  failures.append(traceback.format_exc())
 finally:
  QApplication.instance().quit()

 window=base_create(*args,**kwargs)
 QTimer.singleShot(700,lambda:exercise(window))
 return window

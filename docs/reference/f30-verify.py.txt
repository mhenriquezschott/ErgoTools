import hashlib, json, subprocess, sys, tempfile, zipfile
from pathlib import Path
root = Path.cwd()
def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
reference = Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
actual = {str(p.relative_to(reference)):digest(p) for p in reference.rglob('*') if p.is_file() and '.git' not in p.relative_to(reference).parts}
assert actual == json.loads((root/'docs/reference/source-manifest.json').read_text())
assert digest(root/'uv.lock') == json.loads((root/'docs/f29-validation.json').read_text())['lock_sha256']
fixture = json.loads((root/'tests/fixtures/assessment_context_reference.json').read_text())
assert all(digest(reference/name) == sha for name,sha in fixture['source_sha256'].items())
wheel = root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'
code = '''
import sys
from pathlib import Path
from dataclasses import replace
sys.path.insert(0, sys.argv[1])
import ergotools
assert Path(ergotools.__file__).is_relative_to(Path(sys.argv[1]))
from ergotools.domain.assessments.context import WorkplaceContext, WorkerContext, AssessmentContext
from ergotools.domain.assessments.document import AssessmentKind
from ergotools.application.assessment_assignment import AssignmentSnapshot, resolve_assessment_assignment
owner = WorkerContext('project', 'worker', WorkplaceContext('p','s','l','station','day'))
context = AssessmentContext(owner, AssessmentKind.LIFFT)
row = AssignmentSnapshot(owner, 1, True, 2, True)
assert resolve_assessment_assignment(context.worker, (row,)) is row
assert resolve_assessment_assignment(replace(owner, project_id='other'), (row,)) is None
assert resolve_assessment_assignment(owner, (replace(row, placement_active=False),)) is None
try:
    resolve_assessment_assignment(owner, (row, row))
except ValueError:
    pass
else:
    raise AssertionError('Duplicate owner accepted')
print('isolated assessment-context wheel smoke passed')
'''
with tempfile.TemporaryDirectory() as directory:
    with zipfile.ZipFile(wheel) as archive:
        archive.extractall(directory)
    result = subprocess.run([sys.executable,'-I','-c',code,directory],cwd=directory,check=True,capture_output=True,text=True)
    print(result.stdout.strip())
coverage = json.loads((root/'artifacts/coverage.json').read_text())
modules = {name:data['summary']['percent_covered'] for name,data in coverage['files'].items() if name.endswith(('/context.py','/assessment_assignment.py'))}
assert len(modules)==2 and all(value==100 for value in modules.values())
evidence = dict(milestone='F30',verified_on='2026-09-30',python=sys.version.split()[0],tests_passed=304,new_tests=11,reference_cases=13,coverage_percent=coverage['totals']['percent_covered'],new_module_coverage=modules,reference_files_unchanged=len(actual),dependency_lock_unchanged=True,lock_sha256=digest(root/'uv.lock'),wheel_sha256=digest(wheel),independent_wheel_smoke='passed',ui_changed=False,schema_changed=False,overall_completion_estimate_percent=55)
(root/'docs/f30-validation.json').write_text(json.dumps(evidence,indent=2)+'\n')
print(json.dumps(evidence,indent=2))

import sys
from pathlib import Path

from PyQt5.QtCore import QCoreApplication, QStandardPaths

from ergotools.bootstrap import main as desktop_main


def main(argv: list[str] | None = None) -> int:
    """Run the integrated desktop with per-user settings or an explicit override.

    Installed console scripts and module execution share this entry point. The
    original development bootstrap flags remain available for diagnostic use.
    Assets are loaded from the installed package, never from the launch folder.
    No dependency installation, project opening or saving occurs here.

    Raises:
        OSError: If Qt cannot locate a writable per-user configuration directory.
    """
    arguments = list(sys.argv[1:] if argv is None else argv)
    QCoreApplication.setApplicationName("ErgoTools")
    QCoreApplication.setOrganizationName("ErgoTools")
    if not any(a == "--settings" or a.startswith("--settings=") for a in arguments):
        directory = QStandardPaths.writableLocation(QStandardPaths.AppConfigLocation)
        if not directory:
            raise OSError("No per-user application configuration directory is available.")
        arguments.extend(("--settings", str(Path(directory) / "settings.json")))
    return desktop_main(["--integrated-preview", *arguments])


if __name__ == "__main__":
    raise SystemExit(main())
''')

from pathlib import Path
from unittest.mock import patch

import pytest

from ergotools.launcher import main


@pytest.mark.parametrize("arguments", [[], ["--disable-vtk"]])
def test_launch_uses_integrated_ui_and_user_config(tmp_path: Path, arguments: list[str]) -> None:
    """Integrated rendering is default and explicit diagnostic options pass through."""
    with (
        patch("ergotools.launcher.QStandardPaths.writableLocation", return_value=str(tmp_path)),
        patch("ergotools.launcher.desktop_main", return_value=7) as desktop,
    ):
        assert main(arguments)==7
    desktop.assert_called_once_with(["--integrated-preview", *arguments, "--settings", str(tmp_path/"settings.json")])
    assert not (tmp_path/"settings.json").exists()


@pytest.mark.parametrize("arguments", [["--settings", "custom.json"], ["--settings=custom.json"]])
def test_explicit_settings_are_preserved(arguments: list[str]) -> None:
    """Do not replace an explicitly selected settings file or mutate caller arguments."""
    before=list(arguments)
    with patch("ergotools.launcher.desktop_main", return_value=0) as desktop:
        main(arguments)
    assert arguments==before
    desktop.assert_called_once_with(["--integrated-preview", *arguments])


def test_unavailable_config_path_fails_before_launch() -> None:
    """No implicit working-directory fallback can scatter preferences beside projects."""
    with (
        patch("ergotools.launcher.QStandardPaths.writableLocation", return_value=""),
        patch("ergotools.launcher.desktop_main") as desktop,
        pytest.raises(OSError,match="configuration directory"),
    ):
        main([])
    desktop.assert_not_called()
''')

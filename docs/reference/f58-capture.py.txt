"""Capture original placement updates using isolated in-memory SQLite only."""
import hashlib
import json
import runpy
import sqlite3
from contextlib import closing
from pathlib import Path

root = Path.cwd()
reference = Path(json.loads((root / 'docs/reference/provenance.json').read_text())['reference_root'])
source = reference / 'src/job_placement_repository.py'
api = runpy.run_path(str(source))
key = api['WorkplaceKey']
cases = []
for selected in (['1'], [], ['1', '2'], ['2', '3']):
    with closing(sqlite3.connect(':memory:')) as conn:
        conn.executescript("""
            CREATE TABLE Job(id TEXT PRIMARY KEY);
            CREATE TABLE Station(plant_name TEXT,section_name TEXT,line_name TEXT,id TEXT);
            CREATE TABLE Shift(id TEXT);
            CREATE TABLE WorkplaceContext(
                id INTEGER PRIMARY KEY,plant_name TEXT,section_name TEXT,line_name TEXT,
                station_id TEXT,shift_id TEXT,
                UNIQUE(plant_name,section_name,line_name,station_id,shift_id));
            CREATE TABLE JobPlacement(
                id INTEGER PRIMARY KEY,job_id TEXT,workplace_context_id INTEGER,
                active INTEGER,valid_from TEXT,valid_to TEXT,notes TEXT);
            INSERT INTO Job VALUES('j'),('k');
            INSERT INTO Shift VALUES('Day');
            INSERT INTO Station VALUES('P','S','L','1'),('P','S','L','2'),('P','S','L','3');
            INSERT INTO WorkplaceContext VALUES
                (1,'P','S','L','1','Day'),(2,'P','S','L','2','Day'),(3,'P','S','L','3','Day');
            INSERT INTO JobPlacement VALUES
                (1,'j',1,1,NULL,NULL,'original'),
                (2,'j',2,0,'2020-01-01','2020-12-31','older'),
                (3,'j',2,0,'2021-01-01','2021-12-31','latest'),
                (4,'k',1,1,NULL,NULL,'other');
        """)
        api['replace_active_job_placements'](conn, 'j', [key('P','S','L',s,'Day') for s in selected])
        rows = conn.execute("""
            SELECT p.id,p.job_id,c.station_id,p.active,p.valid_from,p.valid_to,p.notes
            FROM JobPlacement p JOIN WorkplaceContext c ON c.id=p.workplace_context_id
            ORDER BY p.id
        """).fetchall()
        cases.append(dict(selected=selected, rows=rows))
fixture = dict(source_sha256=hashlib.sha256(source.read_bytes()).hexdigest(),cases=cases)
(root / 'tests/fixtures/job_placements_reference.json').write_text(json.dumps(fixture,indent=2)+'\n')
print('Captured four original placement-selection cases without filesystem databases.')

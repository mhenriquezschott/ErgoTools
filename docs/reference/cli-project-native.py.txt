"""Exercise positional startup and native bilingual Recent Projects on a copy."""
import hashlib
import json
import shutil
import sys
import traceback
from pathlib import Path

from PyQt5.QtCore import QPoint, QTimer, Qt
from PyQt5.QtGui import QImage, QPainter
from PyQt5.QtTest import QTest
from PyQt5.QtWidgets import QApplication, QInputDialog
from vtkmodules.vtkIOImage import vtkPNGWriter
from vtkmodules.vtkRenderingCore import vtkWindowToImageFilter

from ergotools import bootstrap
from ergotools.launcher import main

root = Path.cwd()
out = root / 'artifacts/cli-project-native'
out.mkdir(exist_ok=True)
stage = Path('/tmp/ergotools-cli-project-native')
stage.mkdir(exist_ok=True)
source = stage / 'Comprehensive project with spaces.ergprj'
shutil.copyfile(root / 'artifacts/m10-integration-final/run/M10-validated.ergprj', source)
before = hashlib.sha256(source.read_bytes()).hexdigest()
settings = stage / 'settings.json'
settings.unlink(missing_ok=True)
original_factory = bootstrap.create_window
report = {'checks': {}, 'captures': [], 'failures': []}

def menu_capture(window, name):
    menu = window._assessment_menu
    window.refresh_recent_projects()
    menu.project.popup(window.mapToGlobal(QPoint(5, 40)))
    QTest.qWait(100)
    menu.recent.popup(menu.project.mapToGlobal(QPoint(menu.project.width(), 45)))
    QTest.qWait(200)
    assert menu.recent.grab().save(str(out / f'{name}.png'))
    report['captures'].append(name)
    menu.recent.hide()
    menu.project.hide()

def complete_capture(window, name):
    QTest.qWait(150)
    for component in (window.results_sidebar.risk_gauge, window.results_sidebar.damage_progress):
        if component.has_value:
            component.animation.setCurrentTime(component.animation.duration())
        component.animation.stop()
    window._body_renderer._stop_animation()
    image = QImage(window.size(), QImage.Format_ARGB32_Premultiplied)
    image.fill(Qt.transparent)
    window.render(image, QPoint())
    renderer = window._body_renderer
    renderer.render_window.Render()
    filt = vtkWindowToImageFilter()
    filt.SetInput(renderer.render_window)
    filt.ReadFrontBufferOff()
    filt.Update()
    writer = vtkPNGWriter()
    writer.SetInputConnection(filt.GetOutputPort())
    writer.SetFileName(str(out / 'vtk-frame.png'))
    writer.Write()
    frame = QImage(str(out / 'vtk-frame.png'))
    viewport = window.body_panel.viewport
    rect = viewport.geometry()
    rect.moveTopLeft(viewport.mapTo(window, QPoint()))
    painter = QPainter(image)
    painter.drawImage(rect, frame)
    painter.end()
    assert image.save(str(out / f'{name}.png'))
    report['captures'].append(name)

def finish(window):
    try:
        workspace = window.presenter.workspace
        assert workspace.location == str(source) and workspace.active is not None
        assert window._settings_store.load().recent_documents == (str(source),)
        assert window._body_renderer is not None
        complete_capture(window, '02-cli-open-main-en')
        menu_capture(window, '03-recent-populated-en')
        window.languages.setCurrentIndex(1)
        QTest.qWait(300)
        complete_capture(window, '04-cli-open-main-es')
        menu_capture(window, '05-recent-populated-es')
        assert window._assessment_menu.recent.title() == 'Proyectos recientes'
        assert hashlib.sha256(source.read_bytes()).hexdigest() == before
        report['checks'] = {'positional_spaced_project_opened': True, 'recorded_in_recent': True,
                            'vtk_visible': True, 'english_spanish': True, 'no_automatic_save': True}
    except Exception:
        report['failures'].append(traceback.format_exc())
    finally:
        (out / 'verification.json').write_text(json.dumps(report, indent=2))
        QApplication.instance().quit()

def factory(*args, **kwargs):
    window = original_factory(*args, **kwargs)
    original_choose = window.choose_item
    def choose(*args, **kwargs):
        menu_capture(window, '01-recent-empty-en')
        def accept():
            dialog = QApplication.activeModalWidget()
            assert isinstance(dialog, QInputDialog)
            dialog.accept()
        QTimer.singleShot(200, accept)
        return original_choose(*args, **kwargs)
    window.choose_item = choose
    QTimer.singleShot(1000, lambda: finish(window))
    return window

bootstrap.create_window = factory
code = main([str(source), '--settings', str(settings)])
assert code == 0 and not report['failures'], report
print(json.dumps(report))

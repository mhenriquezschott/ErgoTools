import hashlib,importlib.metadata,json,subprocess,sys,tempfile,zipfile
from pathlib import Path
import xml.etree.ElementTree as ET
import tomllib
import pulp,highspy
root=Path.cwd()
def digest(path):return hashlib.sha256(path.read_bytes()).hexdigest()
manifest=json.loads((root/'docs/reference/source-manifest.json').read_text())
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
assert all(digest(reference/name)==sha for name,sha in manifest.items())
old=json.loads((root/'docs/reference/f120-prior-lock-versions.json').read_text())
new={p['name']:p['version'] for p in tomllib.loads((root/'uv.lock').read_text())['package']}
assert all(new.get(name)==version for name,version in old.items())
assert set(new)-set(old)=={'pulp','highspy'}
assert sys.version.split()[0]=='3.13.14'
assert importlib.metadata.version('pulp')=='3.2.2'
assert importlib.metadata.version('highspy')=='1.11.0'
assert highspy.Highs().version()=='1.11.0'
cbc=Path(pulp.PULP_CBC_CMD().path)
expected=json.loads((root/'docs/reference/f119-jrot-package-review.json').read_text())['pulp']['cbc_binary_sha256']
assert digest(cbc)==expected
version=subprocess.run([str(cbc),'-stop'],check=True,capture_output=True,text=True).stdout
assert 'Version: 2.10.3' in version
wheel=root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'
code='import sys\nsys.path.insert(0,sys.argv[1])\n'+(root/'docs/reference/f120-wheel-checks.py.txt').read_text()
with tempfile.TemporaryDirectory() as directory:
 with zipfile.ZipFile(wheel) as archive:archive.extractall(directory)
 smoke=subprocess.run([sys.executable,'-I','-c',code,directory],cwd=directory,capture_output=True,text=True,timeout=30)
 assert smoke.returncode==0,smoke.stdout+smoke.stderr
 print(smoke.stdout.strip())
coverage=json.loads((root/'artifacts/coverage.json').read_text())
tests=ET.parse(root/'artifacts/tests.xml')
evidence={'milestone':'F120','verified_on':'2026-10-08','python':sys.version.split()[0],'tests_passed':len(list(tests.iter('testcase'))),'new_tests':26,'coverage_percent':coverage['totals']['percent_covered'],'ruff_formatted_files':441,'mypy_source_files':199,'architecture_violations':0,'reference_files_unchanged':len(manifest),'prior_lock_versions_retained':len(old),'approved_lock_additions':{name:new[name] for name in sorted(set(new)-set(old))},'lock_sha256':digest(root/'uv.lock'),'cbc_version':'2.10.3','cbc_binary_sha256':digest(cbc),'highs_version':highspy.Highs().version(),'pulp_distribution_version':importlib.metadata.version('pulp'),'pulp_internal_version_string':pulp.__version__,'wheel_sha256':digest(wheel),'independent_wheel_smoke':'passed; actual CBC and two-stage HiGHS solves','native_regression_gate':'xcb full suite; no UI changes or new visual captures','limitations':['JROT form/persistence/main navigation integration pending','Timeout/no-incumbent regressions inject termination evidence deterministically; no timing-sensitive timeout benchmark claimed','Optimal records backend report under reference 1% gap, not exact global optimality','No Windows/macOS solver packaging acceptance','No reliable whole-app completion percentage or finish date']}
(root/'docs/f120-validation.json').write_text(json.dumps(evidence,indent=2)+'\n')
print('F120: reference unchanged, 33 prior lock versions retained, approved solver identities and wheel verified.')

"""Capture original draft creation/copy/save using in-memory SQLite only."""
import ast
import hashlib
import json
import sqlite3
from collections.abc import Iterable, Mapping
from contextlib import closing
from pathlib import Path

root=Path.cwd()
ref=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
path=ref/'src/job_risk_repository.py'
tree=ast.parse(path.read_text())
names={'create_draft_profile','save_draft_profile'}
nodes=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name in names]
env=dict(sqlite3=sqlite3,Iterable=Iterable,Mapping=Mapping,JobRiskProfileError=ValueError,
         PROFILE_SOURCE_TYPES=('expert','external','study','aggregate','imported'))
exec(compile(ast.Module(body=nodes,type_ignores=[]),str(path),'exec'),env)
cases=[]
for mode in ('blank','copy','edit'):
 with closing(sqlite3.connect(':memory:')) as c:
  c.executescript("""
   CREATE TABLE Job(id TEXT);
   INSERT INTO Job VALUES ('j'),('k');
   CREATE TABLE JobRiskProfile(
    id INTEGER PRIMARY KEY,job_id TEXT,name TEXT,version INTEGER,status TEXT,
    is_current INTEGER,source_type TEXT,source_reference TEXT,methodology TEXT,
    sample_size INTEGER,assessed_on TEXT,notes TEXT,updated_at TEXT);
   CREATE TABLE JobRiskMeasurement(
    profile_id INTEGER,tool_id TEXT,total_cumulative_damage REAL,
    probability_outcome REAL,unit TEXT,notes TEXT);
   INSERT INTO JobRiskProfile VALUES
    (1,'j','Study',1,'approved',1,'study','source','method',0,'2026-10-03','note',NULL),
    (4,'j','Old',4,'retired',0,'external',NULL,NULL,NULL,NULL,NULL,NULL);
   INSERT INTO JobRiskMeasurement VALUES (1,'LiFFT',0,0,'Imperial','evidence');
  """)
  identity=env['create_draft_profile'](c,'j',copy_from_profile_id=None if mode=='blank' else 1)
  if mode=='edit':
   env['save_draft_profile'](c,identity,name='  Revised  ',source_type='external',
    source_reference='',methodology='  ',sample_size=0,assessed_on='',notes='',
    measurements=[
     dict(tool_id='LiFFT',total_cumulative_damage=0,probability_outcome=0,unit='Metric'),
     dict(tool_id='DUET',total_cumulative_damage=None,probability_outcome=30),
     dict(tool_id='ST',total_cumulative_damage=None,probability_outcome=None),
    ])
  row=c.execute("""SELECT id,job_id,name,version,status,is_current,source_type,
    source_reference,methodology,sample_size,assessed_on,notes
    FROM JobRiskProfile WHERE id=?""",(identity,)).fetchone()
  measurements=c.execute("""SELECT tool_id,total_cumulative_damage,probability_outcome,unit,notes
    FROM JobRiskMeasurement WHERE profile_id=? ORDER BY tool_id""",(identity,)).fetchall()
  cases.append(dict(mode=mode,profile=row,measurements=measurements))
(root/'tests/fixtures/job_risk_drafts_reference.json').write_text(json.dumps(
 dict(source_sha256=hashlib.sha256(path.read_bytes()).hexdigest(),cases=cases),indent=2)+'\n')
print('Captured original blank/copied/edited profile records.')

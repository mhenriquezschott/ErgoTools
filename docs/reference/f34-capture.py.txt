import ast, hashlib, json, sqlite3, sys, tempfile
from contextlib import closing
from pathlib import Path
from types import SimpleNamespace
root=Path.cwd()
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])/'src/main.py'
tree=ast.parse(reference.read_text())
methods=['getPlants','getSections','getLines','getStations','getShifts']
nodes=[next(node for node in ast.walk(tree) if isinstance(node,ast.FunctionDef) and node.name==name) for name in methods]
namespace={'sqlite3':sqlite3}
exec(compile(ast.Module(body=nodes,type_ignores=[]),str(reference),'exec'),namespace)
plants=['Z plant','A plant','Empty plant']
sections=[['Z plant','Shared'],['A plant','Shared'],['Z plant','Empty section']]
lines=[['Z plant','Shared','Shared line'],['A plant','Shared','Shared line'],['Z plant','Shared','Empty line']]
stations=[['Z plant','Shared','Shared line','02'],['Z plant','Shared','Shared line','01'],['A plant','Shared','Shared line','01']]
shifts=['Night','Day']
with tempfile.TemporaryDirectory() as directory:
    path=Path(directory)/'hierarchy.sqlite'
    with closing(sqlite3.connect(path)) as connection, connection:
        connection.executescript('CREATE TABLE Plant(name TEXT); CREATE TABLE Section(plant_name TEXT,name TEXT); CREATE TABLE Line(plant_name TEXT,section_name TEXT,name TEXT); CREATE TABLE Station(plant_name TEXT,section_name TEXT,line_name TEXT,id TEXT); CREATE TABLE Shift(id TEXT);')
        for table,rows,width in [('Plant',[(x,) for x in plants],1),('Section',sections,2),('Line',lines,3),('Station',stations,4),('Shift',[(x,) for x in shifts],1)]:
            connection.executemany('INSERT INTO '+table+' VALUES ('+','.join('?' for _ in range(width))+')',rows)
    owner=SimpleNamespace(projectFileCreated=True,projectdatabasePath=str(path))
    queries=[]
    for method,parents in [('getPlants',[[]]),('getSections',[[x] for x in plants]),('getLines',sections),('getStations',lines),('getShifts',[[]])]:
        for args in parents:
            queries.append(dict(method=method,args=args,expected=namespace[method](owner,*args)))
fixture=dict(python=sys.version.split()[0],source_sha256=hashlib.sha256(reference.read_bytes()).hexdigest(),plants=plants,sections=sections,lines=lines,stations=stations,shifts=shifts,queries=queries,notes='Actual AST-extracted reference hierarchy methods on a temporary minimal SQLite database. No indexes or operational data. Captured order is not a universal SQLite guarantee.')
(root/'tests/fixtures/work_hierarchy_reference.json').write_text(json.dumps(fixture,indent=2)+'\n')
print(len(queries),'hierarchy query outputs captured')

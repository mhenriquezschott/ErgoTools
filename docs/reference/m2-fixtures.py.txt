"""Organize immutable source copies and verify relocated native test fixtures."""
import hashlib
import json
import shutil
import tempfile
import xml.etree.ElementTree as ET
from pathlib import Path
from ergotools.infrastructure.assessment_project_file import AssessmentProjectFile
from ergotools.application.session import DocumentKey

r=Path.cwd()
reference=Path(json.loads((r/'docs/reference/provenance.json').read_text())['reference_root'])
originals=r/'test_projects/originals'
originals.mkdir(exist_ok=True)
copied={}
for item in json.loads((r/'artifacts/cleanup-m1r-projects.json').read_text()):
    source=Path(item['source'])
    descriptor=ET.fromstring(source.read_bytes())
    paths={source}
    for tag in ('DataPath','ImagesPath'):
        value=descriptor.findtext(tag)
        if value:
            folder=(source.parent/value).resolve()
            assert folder.is_relative_to(source.parent.resolve())
            if folder.is_dir():
                paths.update(p for p in folder.rglob('*') if p.is_file())
    for path in sorted(paths):
        relative=path.relative_to(reference/'tests')
        target=originals/relative
        target.parent.mkdir(parents=True,exist_ok=True)
        if not target.exists():
            shutil.copy2(path,target)
        digest=hashlib.sha256(path.read_bytes()).hexdigest()
        assert hashlib.sha256(target.read_bytes()).hexdigest()==digest
        copied[str(relative)]=digest
(originals/'README.md').write_text('''# Original project reference copies

These descriptors, databases and image folders are byte-identical copies of the
four supplied reference samples. Original spelling is intentional reference data;
do not rename internal companion paths or use these as editable native projects.
Use the recalculated single-file copies in ../native for supported workflows.
The independent utility can read the two supported originals; the earlier JROT
schema and blank-title test3 remain documented unsupported inputs. No source
repair is implied. Source hashes are in ../../artifacts/cleanup-m2-fixtures.json.
''')
report={'original_copy_files':copied,'native_projects':[]}
for source in sorted((r/'test_projects/native').glob('*.ergprj')):
    digest=hashlib.sha256(source.read_bytes()).hexdigest()
    with AssessmentProjectFile.open(source) as project:
        expected={
            'metadata':project.metadata,
            'hierarchy':project.load_hierarchy(), 'jobs':project.load_jobs(),
            'history':project.load_history(), 'risks':project.load_job_risk(),
            'positions':project.load_plot_positions(), 'schemes':project.load_rotation_schemes(),
        }
        documents=project.list_assessments()
        inputs={entry.document:project.load(entry.document) for entry in documents}
        results={entry.document:project.load_saved_result(entry.document) for entry in documents}
        images={plant:project.load_plant_image(plant) for plant in expected['hierarchy'].plants}
    with tempfile.TemporaryDirectory(prefix='ergotools-m2-native-only-') as temp:
        destination=Path(temp)/'relocated.ergprj'
        shutil.copy2(source,destination)
        assert list(Path(temp).iterdir())==[destination]
        with AssessmentProjectFile.open(destination) as project:
            assert project.metadata==expected['metadata']
            assert project.load_hierarchy()==expected['hierarchy']
            assert project.load_jobs()==expected['jobs']
            assert project.load_history()==expected['history']
            assert project.load_job_risk()==expected['risks']
            assert project.load_plot_positions()==expected['positions']
            assert project.load_rotation_schemes()==expected['schemes']
            assert all(project.load(key)==value for key,value in inputs.items())
            assert all(project.load_saved_result(key)==value for key,value in results.items())
            assert all(project.load_plant_image(plant)==image for plant,image in images.items())
    assert hashlib.sha256(source.read_bytes()).hexdigest()==digest
    report['native_projects'].append({'path':str(source.relative_to(r)),'sha256':digest,
        'assessment_versions':len(inputs),'results':len(results),
        'plants':len(images),'embedded_images':sum(v is not None for v in images.values()),
        'rotation_schemes':len(expected['schemes']),'relocated_catalogs_sources_results_images_schemes_equal':True})
manifest=json.loads((r/'docs/reference/source-manifest.json').read_text())
assert all(hashlib.sha256((reference/p).read_bytes()).hexdigest()==v for p,v in manifest.items())
report['reference_files_unchanged']=len(manifest)
report['lock_sha256']=hashlib.sha256((r/'uv.lock').read_bytes()).hexdigest()
(r/'artifacts/cleanup-m2-fixtures.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'original_files':len(copied),'native_projects':report['native_projects']}))

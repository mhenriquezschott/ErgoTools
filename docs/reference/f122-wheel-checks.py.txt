import sys
from pathlib import Path
from PyQt5.QtCore import Qt
from PyQt5.QtTest import QTest
from PyQt5.QtWidgets import QApplication,QDialog
from ergotools.bootstrap import load_icons
from ergotools.application.rotation_scope import RotationScopeOptions
from ergotools.domain.assessments.context import WorkplaceContext
from ergotools.domain.language import Language
from ergotools.presentation.desktop.rotation_scope_dialog import RotationScopeDialog
app=QApplication.instance() or QApplication([])
contexts=tuple(WorkplaceContext('P','S','L',station,'Day') for station in ('1','2'))
options=RotationScopeOptions(('Day',),contexts)
for language in Language:
 dialog=RotationScopeDialog(options,load_icons(),language=language)
 dialog.resize(680,560);dialog.show();app.processEvents()
 QTest.mouseClick(dialog.use_button,Qt.LeftButton)
 assert dialog.isVisible() and dialog.result()==QDialog.Rejected
 plant=dialog.tree.topLevelItem(0);plant.setCheckState(0,Qt.Checked)
 assert dialog.selected_contexts()==contexts
 if language==Language.SPANISH:assert dialog.selection_label.text()=='2 estaciones seleccionadas'
 QTest.mouseClick(dialog.use_button,Qt.LeftButton)
 assert dialog.result()==QDialog.Accepted
 dialog.deleteLater();app.processEvents()
 cancel=RotationScopeDialog(options,load_icons(),selected=(contexts[0],),language=language)
 cancel.show();app.processEvents();QTest.keyClick(cancel,Qt.Key_Escape)
 assert cancel.result()==QDialog.Rejected
 cancel.deleteLater();app.processEvents()
for name,module in tuple(sys.modules.items()):
 if name.startswith('ergotools') and getattr(module,'__file__',None):
  assert Path(module.__file__).resolve().is_relative_to(Path(sys.argv[1]).resolve()),name
print('Extracted-wheel native EN/ES original scope actions, empty-selection protection, exact identities and Escape passed; all ErgoTools imports came from extraction.')

import ast,json
from pathlib import Path
root=Path.cwd();ref=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root']);modules={p.stem:p for p in (ref/'src').glob('*.py')};trees={name:ast.parse(p.read_text()) for name,p in modules.items()}
imports={name:sorted({(n.module or '').split('.')[0] for n in ast.walk(tree) if isinstance(n,ast.ImportFrom)} | {alias.name.split('.')[0] for n in ast.walk(tree) if isinstance(n,ast.Import) for alias in n.names}) for name,tree in trees.items()}
reachable=set();todo=['main']
while todo:
 name=todo.pop()
 if name in reachable:continue
 reachable.add(name);todo.extend(i for i in imports[name] if i in modules and i not in reachable)
report=[]
for name,tree in trees.items():
 connections=[];overrides=[]
 for node in ast.walk(tree):
  if isinstance(node,ast.Call) and isinstance(node.func,ast.Attribute) and node.func.attr=='connect':
   connections.append({'line':node.lineno,'signal':ast.unparse(node.func.value),'callbacks':[ast.unparse(a) for a in node.args]})
  if isinstance(node,ast.FunctionDef) and (node.name.endswith('Event') or node.name in ('paint','boundingRect','shape')):overrides.append({'line':node.lineno,'method':node.name})
 report.append({'module':name,'source':'src/'+modules[name].name,'reachable_import_from_main':name in reachable,'imports':[i for i in imports[name] if i in modules],'connections':connections,'qt_overrides':overrides})
public=[];missing=[]
for p in sorted((root/'src/ergotools').rglob('*.py')):
 tree=ast.parse(p.read_text())
 def visit(body,prefix=''):
  for n in body:
   if isinstance(n,(ast.ClassDef,ast.FunctionDef,ast.AsyncFunctionDef)):
    if not n.name.startswith('_'):
     item={'file':str(p.relative_to(root)),'line':n.lineno,'symbol':prefix+n.name,'signature':ast.unparse(n).split('\n')[0],'doc':ast.get_docstring(n)}
     public.append(item)
     if not item['doc']:missing.append(item)
    if isinstance(n,ast.ClassDef):visit(n.body,prefix+n.name+'.')
 visit(tree.body)
out=root/'artifacts/m11-audit';out.mkdir(exist_ok=True)
(out/'original-entry-points.json').write_text(json.dumps(report,indent=2)+'\n');(out/'public-api.json').write_text(json.dumps(public,indent=2)+'\n')
print('Original modules',len(report),'reachable imports',len(reachable),'connections',sum(len(m['connections']) for m in report),'public APIs',len(public),'missing docs',len(missing))
for m in report:print(m['module'], 'reachable' if m['reachable_import_from_main'] else 'not imported',len(m['connections']),len(m['qt_overrides']))
print('Missing',missing)
ledger=root/'docs/finish-progress.json';l=json.loads(ledger.read_text());next(m for m in l['milestones'] if m['id']=='M11')['status']='in_progress';l['implementation_status']='M11_in_progress; M12_not_started';l['batch_authorization']='Owner authorizes M11 only on 2026-10-11.';ledger.write_text(json.dumps(l,indent=2)+'\n')

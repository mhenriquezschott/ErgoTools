import hashlib,json,shutil,sys,tempfile
from pathlib import Path
from ergotools.infrastructure.legacy_project import read_legacy_descriptor,LegacyProject
root=Path.cwd();ref=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
source=ref/'tests/ErgoTools_ComparativeRiskTest.ergprj';original=read_legacy_descriptor(source,'validation')
with tempfile.TemporaryDirectory() as directory:
 target=Path(directory)/source.name;shutil.copyfile(source,target)
 descriptor=read_legacy_descriptor(target,'validation');descriptor.database_path.parent.mkdir(parents=True);shutil.copyfile(original.database_path,descriptor.database_path)
 before=hashlib.sha256(descriptor.database_path.read_bytes()).hexdigest()
 with LegacyProject(descriptor) as project:
  inventory=project.inventory();assert inventory.schema_version==8
  assert dict(inventory.table_counts)['Worker']==28 and dict(inventory.table_counts)['IndividualAssessment']==191
  from ergotools.infrastructure.legacy_catalogs import read_legacy_catalogs
  catalogs=read_legacy_catalogs(project)
  assert len(catalogs.hierarchy.work.workers)==28 and len(catalogs.jobs.jobs)==12
  assert len(catalogs.jobs.assignments)==104 and len(catalogs.risks.profiles)==19
  assert sum(len(profile.measurements) for profile in catalogs.risks.profiles)==56
  from ergotools.infrastructure.legacy_assessments import read_legacy_assessments,LegacyShoulderSource
  assessments=read_legacy_assessments(project,catalogs)
  assert len(assessments)==191 and sum(len(a.source.tasks) for a in assessments)==3532
  assert sum(isinstance(a.source,LegacyShoulderSource) for a in assessments)==56
  source_ids={r['id'] for r in project.rows('IndividualAssessment')}
  assert {a.source_id for a in assessments}==source_ids
  assert all(len(a.original_tasks)==len(a.task_evidence)==len(a.source.tasks) for a in assessments)
  assert sum(raw.get('repetitions')=='' for a in assessments for raw in a.original_tasks)==1358
  from ergotools.infrastructure.legacy_reconciliation import reconcile_legacy_assessments
  plan=reconcile_legacy_assessments(project,catalogs,assessments)
  assert plan.matched_summaries==191 and len(plan.recovered)==3 and not plan.conflicts
  assert sum(len(a.source.tasks) for a in plan.recovered)==45
  assert {a.ownership.assignment_id for a in plan.recovered}=={101,102,104}
  assert len({a.ownership.document_id for a in plan.recovered})==3
 assert hashlib.sha256(descriptor.database_path.read_bytes()).hexdigest()==before

record={'checkpoint':'F131','matched_summaries':plan.matched_summaries,'recovered_assessments':len(plan.recovered),'recovered_task_rows':sum(len(a.source.tasks) for a in plan.recovered),'recovered_assignment_ids':[a.ownership.assignment_id for a in plan.recovered],'conflicts':len(plan.conflicts),'policy':'No source writes, no inferred reassignment, stable full-context/tool document IDs; all old and normalized task/cache evidence checked.'}
(root/'docs/f131-legacy-audit.json').write_text(json.dumps(record,indent=2)+'\n')
print(record)

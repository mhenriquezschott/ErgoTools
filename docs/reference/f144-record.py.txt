import hashlib,json,re,shutil,sys,xml.etree.ElementTree as ET
from pathlib import Path
root=Path.cwd();checkpoint=sys.argv[1];label=checkpoint.upper()
def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
manifest=json.loads((root/'docs/reference/source-manifest.json').read_text());reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
assert len(manifest)==500 and all(digest(reference/name)==sha for name,sha in manifest.items())
assert digest(root/'uv.lock')==json.loads((root/'docs/f127-validation.json').read_text())['lock_sha256']
xml=ET.parse(root/'artifacts/tests.xml');cases=list(xml.iter('testcase'));assert not list(xml.iter('failure')) and not list(xml.iter('error'))
coverage=json.loads((root/'artifacts/coverage.json').read_text())['totals']['percent_covered'];log=(root/'artifacts'/(checkpoint+'-check.log')).read_text();assert str(len(cases))+' passed' in log
smoke=(root/'artifacts'/(checkpoint+'-wheel.log')).read_text();assert 'passed' in smoke
note=Path('/tmp/'+checkpoint+'-notes.md').read_text()
record={'checkpoint':label,'date':'2026-10-10','tests_passed':len(cases),'coverage_percent':coverage,'reference_files_unchanged':500,'lock_sha256':digest(root/'uv.lock'),'wheel_sha256':digest(root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'),'independent_wheel':smoke.strip(),'ui_changes':False,'visual_validation':'No UI changes; no new visual parity claim. F127 integrated UI evidence retained.','limits':['Legacy adapters read version 8 only; destination conversion and comprehensive integrated UI validation remain.','Overall completion percentage/finish date unvalidated.']}
(root/'docs'/(checkpoint+'-validation.json')).write_text(json.dumps(record,indent=2)+'\n')
(root/'docs'/(checkpoint+'-legacy-compatibility.md')).write_text(note+f'\nValidation: {len(cases)} tests pass; {coverage:.2f}% coverage. Ruff/format, strict mypy, architecture and builds pass. Extracted-wheel compatibility checks pass. All 500 reference hashes and the dependency lock are unchanged. No UI change. Evidence: {checkpoint}-validation.json and artifacts/{checkpoint}-*.log.\n')
for name in ['wheel-checks','audit']:
 source=Path('/tmp/'+checkpoint+'-'+name+'.py')
 if source.exists():shutil.copyfile(source,root/'docs/reference'/(checkpoint+'-'+name+'.py.txt'))
shutil.copyfile('/tmp/record-legacy-checkpoint.py',root/'docs/reference'/(checkpoint+'-record.py.txt'))
status=root/'docs/STATUS.md';s=status.read_text();match=re.search(r'## Latest checkpoint: (F\d+)',s);previous=match[1];s=s.replace(match[0],'## '+previous+' historical checkpoint',1)
index=s.index('## '+previous+' historical checkpoint')
summary=note.split('\n\n',2)[1]
s=s[:index]+f'## Latest checkpoint: {label}\n\n'+summary+f'\n\n{len(cases)} tests pass, {coverage:.2f}% coverage; required gates, extracted wheel and reference/lock integrity pass. See [{label} compatibility evidence]({checkpoint}-legacy-compatibility.md). No UI changes.\n\n'+s[index:];status.write_text(s)
for filename in ['architecture.md','development.md']:
 p=root/'docs'/filename;p.write_text(p.read_text()+f'\n\n## {label} legacy compatibility\n\n'+summary+f' See {checkpoint}-legacy-compatibility.md and {checkpoint}-validation.json for mapping, checks and remaining limits.\n')
print(label,len(cases),'tests',round(coverage,2),'coverage; checkpoint recorded.')

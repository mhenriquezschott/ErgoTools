"""Verify native project values, wheel separation and all reference/lock hashes."""
import hashlib
import json
import subprocess
import sys
import zipfile
from pathlib import Path

root=Path.cwd()
stage=Path('/tmp/ergotools-cleanup-m1r')
stage.mkdir(exist_ok=True)
wheel_root=stage/'wheel'
wheel_root.mkdir(exist_ok=True)
wheel=root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'
with zipfile.ZipFile(wheel) as archive:
    assert not any(name.startswith('tools/') or 'legacy_' in name or 'recorded_result_store' in name or 'archived_results' in name for name in archive.namelist())
    archive.extractall(wheel_root)
    sources=list((root/'src/ergotools').rglob('*.py'))
    for path in sources:
        assert archive.read('ergotools/'+str(path.relative_to(root/'src/ergotools')))==path.read_bytes()
previous=root/'artifacts/m10-integration-final/run/M10-validated.ergprj'
original_hash=hashlib.sha256(previous.read_bytes()).hexdigest()
upgraded=stage/'upgraded previous.ergprj'
subprocess.run([sys.executable,'tools/project_migration/migrate_project.py',str(previous),'--output',str(upgraded)],check=True)
assert hashlib.sha256(previous.read_bytes()).hexdigest()==original_hash
native=root/'test_projects/native/comparative_risk_test.ergprj'
subprocess.run(['xvfb-run','-a','-s','-screen 0 1920x1200x24',sys.executable,'-I',
                '/tmp/cleanup_m1r_native.py',str(wheel_root),str(native),str(root/'artifacts/cleanup-m1r-native')],
               cwd=stage,check=True,timeout=90)
# Validate both regenerated fixtures and the previous converted acceptance project.
from ergotools.application.saved_results import calculate_saved_result
from ergotools.application.session import DocumentKey
from ergotools.infrastructure.assessment_project_file import AssessmentProjectFile
from ergotools.application.assessment_export import saved_summary_rows
from ergotools.domain.assessments.document import AssessmentKind
from ergotools.presentation.summary_csv import summary_csv_table
import sqlite3
from contextlib import closing
report=[]
for path in (native,root/'test_projects/native/integrated_test.ergprj',upgraded):
    with AssessmentProjectFile.open(path) as project:
        statuses={}
        for entry in project.load_history().entries:
            key=DocumentKey(project.metadata.project_id,entry.document_id)
            result=project.load_saved_result(key)
            expected=calculate_saved_result(key,project.load(key),project.metadata.default_units)
            assert result==expected
            status='unavailable' if result is None else result.calculation_status.value
            statuses[status]=statuses.get(status,0)+1
        for kind in AssessmentKind:
            rows=saved_summary_rows(project,kind)
            data=summary_csv_table(rows,'fixed')[1:]
            for row,csv in zip(rows,data,strict=True):
                if row.total_damage is not None:
                    assert float(csv[8])==row.total_damage
                if row.recorded_probability_percent is not None:
                    assert float(csv[9])==row.recorded_probability_percent
        images=sum(project.load_plant_image(plant) is not None for plant in project.load_hierarchy().plants)
        rotations=len(project.load_rotation_schemes())
    with closing(sqlite3.connect(path)) as connection:
        tables={row[0] for row in connection.execute("SELECT name FROM sqlite_master WHERE type='table'")}
        assert 'recorded_assessment_results' not in tables
        assert not any(name.startswith('legacy_') for name in tables)
        count=connection.execute('SELECT count(*) FROM saved_assessment_results').fetchone()[0]
        assert count==sum(value for key,value in statuses.items() if key!='unavailable')
    report.append({'path':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
                   'calculation_statuses':statuses,'sole_result_rows':count,'embedded_images':images,'schemes':rotations})
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
manifest=json.loads((root/'docs/reference/source-manifest.json').read_text())
assert all(hashlib.sha256((reference/path).read_bytes()).hexdigest()==digest for path,digest in manifest.items())
lock=hashlib.sha256((root/'uv.lock').read_bytes()).hexdigest()
assert lock=='9e1f2f9f811ec3e9f57f435bd719d7e66112fb4977f6a29f0f845ce4dcaf39ae'
(root/'artifacts/cleanup-m1r-package.json').write_text(json.dumps({
    'wheel_sha256':hashlib.sha256(wheel.read_bytes()).hexdigest(),
    'source_python_files_equal':len(sources),'wheel_excludes_converter_and_duplicate_store':True,
    'native_workflows_with_converter_imports_blocked':True,'native_projects':report,
    'previous_source_hash_unchanged':original_hash,'reference_files_unchanged':len(manifest),'lock_sha256':lock,
},indent=2)+'\n')
print(json.dumps(report,indent=2))

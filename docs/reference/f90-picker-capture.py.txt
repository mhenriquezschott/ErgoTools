
from pathlib import Path
import sys,tempfile
from PyQt5.QtWidgets import QApplication,QLineEdit
from PyQt5.QtCore import QTimer
from ergotools.bootstrap import create_window
app=QApplication([]);out=Path.cwd()/'artifacts/project-snapshots'
title=sys.argv[1]
with tempfile.TemporaryDirectory() as directory:
 window=create_window(Path(directory)/'settings.json',original_navigation=True,enable_vtk=True)
 window.show()
 def select():
  modal=QApplication.activeModalWidget()
  modal.setDirectory(directory)
  modal.findChild(QLineEdit,'fileNameEdit').setText('copy' if title=='save_as' else 'backup')
  def finish():
   modal.grab().save(str(out/(title.replace('_','-')+'-picker.png')))
   modal.reject() if title=='save_as' else modal.accept()
  QTimer.singleShot(100,finish)
 QTimer.singleShot(250,select)
 location=window.choose_file(True,title_key=title)
 if title=='save_as':assert location==''
 else:assert location==str(Path(directory)/'backup.ergprj')
 window.close();app.processEvents();assert window._body_renderer._closed
 print(title,'native selected picker passed',flush=True)

import hashlib,json,subprocess,sys,tempfile,zipfile,xml.etree.ElementTree as ET
from pathlib import Path
root=Path.cwd()
def digest(path):return hashlib.sha256(path.read_bytes()).hexdigest()
manifest=json.loads((root/'docs/reference/source-manifest.json').read_text());reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
assert all(digest(reference/name)==value for name,value in manifest.items())
assert digest(root/'uv.lock')==json.loads((root/'docs/f78-validation.json').read_text())['lock_sha256']
wheel=root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'
code='''
import sys
from pathlib import Path
sys.path.insert(0,sys.argv[1])
from PyQt5.QtWidgets import QApplication
from ergotools.bootstrap import create_window,load_icons
from ergotools.application.document_lifecycle import CloseChoice
from ergotools.domain.project import ProjectMetadata
from ergotools.domain.model_units import MeasurementSystem
from ergotools.domain.assessments.duet_draft import DuetDraft,DuetTask
from ergotools.presentation.desktop.duet_presenter import DuetPresenter
from ergotools.presentation.desktop.duet_task_panel import DuetTaskPanel
from ergotools.infrastructure.assessment_project_file import AssessmentProjectFile
from ergotools.application.assessments.evaluation import EvaluationStatus
app=QApplication([]);harness=create_window(Path('settings.json'));workspace=harness.presenter.workspace
workspace.create(str(Path('duet.ergprj').absolute()),ProjectMetadata('p','DUET',MeasurementSystem.IMPERIAL),'d',DuetDraft((DuetTask('a',5,100),DuetTask('b',0,None))))
panel=DuetTaskPanel(DuetPresenter(workspace),load_icons(),report_issue=lambda issue:None,changed=lambda:None)
panel.load_source(panel.presenter.bind());panel.show();app.processEvents()
panel.inputs[0][0].setCurrentIndex(6);panel.calculate_button.click()

assert panel.outputs[0][0].text()!='0.0'
assert panel.presenter.source().tasks[1].repetitions is None

with AssessmentProjectFile.open(Path('duet.ergprj')) as project:assert project.load(document)==source
panel.reset_button.click();assert panel.presenter.source().tasks[0].omni_res_scale==0
workspace.active.undo();assert workspace.active.state is source
workspace.close(choice=CloseChoice.DISCARD);panel.close();harness.close()
print('Extracted wheel DUET edit/partial calculation/save/reopen/reset/undo and packaged icons passed')
'''
with tempfile.TemporaryDirectory() as directory:
 with zipfile.ZipFile(wheel) as archive:archive.extractall(directory)
 result=subprocess.run([sys.executable,'-I','-c',code,directory],cwd=directory,capture_output=True,text=True)
 if result.returncode:raise RuntimeError(result.stderr)
 print(result.stdout.strip())

evidence['native_screenshots']={str(p.relative_to(root)):digest(p) for p in sorted((root/'artifacts/duet-task-panel').glob('*.png'))}
(root/'docs/f79-validation.json').write_text(json.dumps(evidence,indent=2)+'\n');print('Verified reference hashes, unchanged lock and F79 package evidence')

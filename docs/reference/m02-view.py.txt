  dialog=WorkerManagementDialog(workspace,load_icons(),system,selected_worker='w',language=language);dialog.resize(1360,760);dialog.show();dialog.optional_toggle.click()

  dialog.tabs.setCurrentIndex(1);capture(dialog,prefix+'-assignments')

   w=next(w for w in project.load_hierarchy().work.workers if w.worker_id=='w');assert w.first_name=='Ana' and w.details.hiring_date=='2000-06-08'

 # A project with no workers uses the same form, without an artificial selection.

 dialog=WorkerManagementDialog(workspace,load_icons(),MeasurementSystem.METRIC);dialog.resize(1360,760);dialog.show();capture(dialog,'empty-project');assert dialog.worker_table.rowCount()==0 and not dialog.selected_worker_id;dialog.close();workspace.close(choice=CloseChoice.DISCARD);main.close()
(out/'verification.json').write_text(json.dumps(report,indent=2)+'\n')

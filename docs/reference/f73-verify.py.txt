import hashlib,json,subprocess,sys,tempfile,zipfile,xml.etree.ElementTree as ET
from pathlib import Path
root=Path.cwd()
def digest(path):return hashlib.sha256(path.read_bytes()).hexdigest()
manifest=json.loads((root/'docs/reference/source-manifest.json').read_text())
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
assert all(digest(reference/name)==value for name,value in manifest.items())
assert digest(root/'uv.lock')==json.loads((root/'docs/f71-validation.json').read_text())['lock_sha256']
wheel=root/'dist/ergotools-0.1.0.dev0-py3-none-any.whl'
code='''
import sys
from pathlib import Path
sys.path.insert(0,sys.argv[1])
from PyQt5.QtWidgets import QApplication,QLineEdit
from PyQt5.QtCore import QTimer
from ergotools.bootstrap import create_window
from ergotools.application.document_lifecycle import CloseChoice
from ergotools.domain.project import ProjectMetadata
from ergotools.domain.model_units import MeasurementSystem
from ergotools.domain.assessments.lifft_draft import LifftDraft
from ergotools.presentation.desktop.organization_dialog import OrganizationDialog
from ergotools.infrastructure.assessment_project_file import AssessmentProjectFile
app=QApplication([])
path=Path('organization.ergprj')
window=create_window(Path('settings.json'))
workspace=window.presenter.workspace
workspace.create(str(path),ProjectMetadata('wheel','Wheel',MeasurementSystem.METRIC),'initial',LifftDraft(MeasurementSystem.METRIC))
window.refresh()
completed=[]
def enter_organization():
 dialog=QApplication.activeModalWidget()
 assert isinstance(dialog,OrganizationDialog)
 def save(identity):
  editor=next(iter(dialog.editors.values()))
  assert isinstance(editor,QLineEdit)
  editor.setText(identity)
  dialog.save_button.click()
  assert dialog.current_keys[-1]==identity
 dialog.editors['area'].setText('17.5')
 save('Plant')
 for identity in ('Section','Line','Station'):
  dialog.add_child_button.click()
  save(identity)
 dialog.directory_tabs.setCurrentIndex(1)
 dialog.add_root_button.click()
 save('Day')
 completed.append(True)
 dialog.close()
QTimer.singleShot(0,enter_organization)
window.buttons['organization_management'].click()
assert completed==[True]
workspace.close(choice=CloseChoice.DISCARD)
window.close()
with AssessmentProjectFile.open(path) as project:
 assert project.format_version==7
 hierarchy=project.load_hierarchy()
 assert hierarchy.plants==('Plant',)
 assert hierarchy.work.shifts==('Day',)
 assert dict(hierarchy.organization[0].attributes)['area']==17.5
print('Isolated wheel organization modal/create/save/reopen passed')
'''
with tempfile.TemporaryDirectory() as directory:
 with zipfile.ZipFile(wheel) as archive:archive.extractall(directory)
 result=subprocess.run([sys.executable,'-I','-c',code,directory],cwd=directory,capture_output=True,text=True)
 if result.returncode:raise RuntimeError(result.stderr)
 print(result.stdout.strip())
coverage=json.loads((root/'artifacts/coverage.json').read_text())
evidence=dict(milestone='F73',verified_on='2026-10-04',python=sys.version.split()[0],tests_passed=sum(1 for _ in ET.parse(root/'artifacts/tests.xml').iter('testcase')),new_tests=4,coverage_percent=coverage['totals']['percent_covered'],reference_files_unchanged=len(manifest),dependency_lock_unchanged=True,lock_sha256=digest(root/'uv.lock'),wheel_sha256=digest(wheel),independent_wheel_smoke='passed',hierarchy_payload_version=3,reads_hierarchy_payload_versions=[1,2,3],project_sql_schema_changed=False,overall_completion_estimate_percent=None)
evidence['desktop_new_project_format']=7
evidence['native_screenshots']={str(p.relative_to(root)):digest(p) for p in sorted((root/'artifacts/organization-management').glob('*.png'))}
(root/'docs/f73-validation.json').write_text(json.dumps(evidence,indent=2)+'\n')
print(json.dumps(evidence,indent=2))

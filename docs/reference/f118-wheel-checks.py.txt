from PyQt5.QtWidgets import QApplication,QMessageBox
from PyQt5.QtCore import QTimer
from ergotools.bootstrap import create_window
from ergotools.application.document_lifecycle import CloseChoice
from ergotools.domain.language import Language
from ergotools.domain.assessments.document import AssessmentKind
from ergotools.presentation.desktop.plot_window import PlotWindow
from ergotools.presentation.plot_csv import PLOT_HEADERS
import csv
app=QApplication.instance() or QApplication([])
with tempfile.TemporaryDirectory() as directory:
 fixture=project(Path(directory));repo=next(fixture);location=repo.location if hasattr(repo,'location') else str(Path(directory)/'plot.ergprj')
 fixture.close()
 for language in Language:
  main=create_window(Path(directory)/(language.value+'-settings.json'),original_navigation=True)
  main.presenter.workspace.switch(str(Path(directory)/'plot.ergprj'),'lifft-1');main.language=language;main.translate();main.refresh();main.show();app.processEvents()
  repository=main.presenter.workspace.plot_repository();before=repository.load_history(),repository.load_plot_positions()
  visited=[];failures=[];timer=QTimer(main);timer.setSingleShot(True)
  def interact():
   view=app.activeModalWidget()
   try:
    assert isinstance(view,PlotWindow) and view.repository is repository;visited.append(view)
    view.filters.tools.select_tool(AssessmentKind.DUET);view.filters.apply_filters()
    destination=str(Path(directory)/(language.value+'.csv'));view.choose_csv=lambda:destination
    child=QTimer(view);child.setSingleShot(True)
    def success():
     box=app.activeModalWidget();assert isinstance(box,QMessageBox);box.done(QMessageBox.Ok)
    child.timeout.connect(success);child.start(0);view.map_panel.buttons['export'].click();child.stop()
    with open(destination,newline='',encoding='utf-8') as stream:table=list(csv.reader(stream))
    assert table[0]==list(PLOT_HEADERS) and len(table)>1
    row=next(r for r in view.layer.results.workers if r.worker.worker_id=='x');view.layer.worker_menu(row.assessment.assignment_id).actions()[2].trigger()
    assert not view.isVisible()
   except BaseException as error:
    failures.append(error)
    if isinstance(view,PlotWindow):view.reject()
  timer.timeout.connect(interact);timer.start(0);assert main.main_navigation.toolbar_actions['plot'].isEnabled();main.main_navigation.toolbar_actions['plot'].trigger();timer.stop()
  assert visited and not failures,failures
  assert main.duet_active and main.presenter.workspace.active.document.document_id=='duet-3'
  assert (repository.load_history(),repository.load_plot_positions())==before
  main.presenter.workspace.close(choice=CloseChoice.DISCARD);main.close();main.deleteLater();app.processEvents()

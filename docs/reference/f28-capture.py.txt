from pathlib import Path
import tempfile
from PyQt5.QtWidgets import QApplication
from PyQt5.QtCore import Qt
from PyQt5.QtTest import QTest
from ergotools.bootstrap import create_window
from ergotools.domain.project import ProjectMetadata
from ergotools.domain.model_units import MeasurementSystem
from ergotools.domain.assessments.lifft_draft import LifftDraft,LifftTask
from ergotools.application.document_lifecycle import CloseChoice
app=QApplication([])
out=Path.cwd()/'artifacts/desktop';out.mkdir(exist_ok=True)
with tempfile.TemporaryDirectory() as directory:
 root=Path(directory);w=create_window(root/'settings.json');w.show();app.processEvents()
 w.grab().save(str(out/'native-empty.png'))
 workspace=w.presenter.workspace
 workspace.create(str(root/'project.sqlite'),ProjectMetadata('p','Preview',MeasurementSystem.IMPERIAL),'lifft',LifftDraft(MeasurementSystem.IMPERIAL,tuple(LifftTask(str(i),12+i,10+i,100+i) for i in range(25))))
 w.refresh();w.calculate();app.processEvents();w.grab().save(str(out/'native-populated.png'))
 w.languages.setCurrentIndex(1);w.units.setCurrentIndex(1);w.decimal.setCurrentIndex(1);w.resize(960,680);app.processEvents();w.grab().save(str(out/'spanish-small.png'))
 w.units.showPopup();app.processEvents();app.activePopupWidget().grab().save(str(out/'spanish-expanded.png'));w.units.hidePopup()
 w.table.scrollToBottom();w.table.selectRow(24);w.field.setCurrentIndex(1);w.entry.setFocus();w.entry.selectAll();QTest.keyClicks(w.entry,'invalid');QTest.mouseClick(w.buttons['apply'],Qt.LeftButton);app.processEvents();w.grab().save(str(out/'spanish-invalid.png'))
 w.discard_edit();w.pending=False;workspace.close(choice=CloseChoice.DISCARD)
 workspace.create(str(root/'metric.sqlite'),ProjectMetadata('m','Metric',MeasurementSystem.METRIC),'lifft',LifftDraft(MeasurementSystem.METRIC,(LifftTask('a',30.48,44.5,100),)))
 w.refresh();w.calculate();app.processEvents();w.grab().save(str(out/'metric-review.png'))
 workspace.close(choice=CloseChoice.DISCARD);w.close()
 print('Six native screenshots captured; temporary projects closed.')

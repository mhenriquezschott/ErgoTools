"""Compare migrated outputs against original calculators executed only in memory."""
import json
from pathlib import Path
from ergotools.infrastructure.assessment_project_file import AssessmentProjectFile
from ergotools.application.session import DocumentKey
from ergotools.domain.assessments.duet_draft import DuetDraft
from ergotools.domain.assessments.lifft_draft import LifftDraft
from tools.project_migration.ergotools_migration.legacy_project import LegacyProject,read_legacy_descriptor
from tools.project_migration.ergotools_migration.legacy_catalogs import read_legacy_catalogs
from tools.project_migration.ergotools_migration.legacy_assessments import read_legacy_assessments
from tools.project_migration.ergotools_migration.legacy_reconciliation import reconcile_legacy_assessments

root=Path.cwd()
reference=Path(json.loads((root/'docs/reference/provenance.json').read_text())['reference_root'])
original={}
for module,cls in (('pyLiFFT','LiFFT'),('pyDUET','DUET'),('pyTST','TST')):
 namespace={'__name__':'m1r_original_'+module}
 exec(compile((reference/'src'/f'{module}.py').read_text(),module,'exec'),namespace)
 original[cls]=namespace[cls]
report=[]
for filename,native in (('ErgoTools_ComparativeRiskTest.ergprj','comparative_risk_test.ergprj'),
                        ('ErgoTools_IntegratedTest.ergprj','integrated_test.ergprj')):
 totals={}; tasks_checked=0
 with AssessmentProjectFile.open(root/'test_projects/native'/native) as project:
  for entry in project.load_history().entries:
   key=DocumentKey(project.metadata.project_id,entry.document_id)
   source=project.load(key); result=project.load_saved_result(key); total=0.0; calculator=None
   for task,row in zip(source.tasks,result.tasks,strict=True):
    if isinstance(source,DuetDraft):
     scale=task.omni_res_scale if task.omni_res_scale is not None else 0
     reps=task.repetitions if task.omni_res_scale is not None and task.repetitions is not None else 0
     calculator=original['DUET'](scale,reps)
     damage=calculator.calculate()[0]; moment=None
    else:
     evaluated=task if isinstance(source,LifftDraft) else source.model_task(task.task_id)
     present=evaluated.lever_arm is not None and evaluated.load is not None
     if not isinstance(source,LifftDraft): present=present and evaluated.direction is not None
     arm=evaluated.lever_arm if present else 0
     load=evaluated.load if present else 0
     reps=evaluated.repetitions if present and evaluated.repetitions is not None else 0
     calculator=(original['LiFFT'](source.system.value,arm,load,reps) if isinstance(source,LifftDraft)
                 else original['TST'](source.system.value,evaluated.direction.value if present else '0',arm,load,reps))
     moment,damage,_=calculator.calculate()
    assert row.damage==damage,(native,entry.document_id,task.task_id,'damage')
    assert row.moment==moment,(native,entry.document_id,task.task_id,'moment')
    total+=damage; tasks_checked+=1
   assert result.damage==total,(native,entry.document_id,'total')
   percent=None if total==0 else calculator.riskFromDamage(total)*100
   assert result.probability_percent==percent,(native,entry.document_id,'percent')
   totals[entry.kind.value]=totals.get(entry.kind.value,0)+1
 descriptor=read_legacy_descriptor(reference/'tests'/filename,'comparison')
 mismatches={}
 with LegacyProject(descriptor) as legacy:
  catalogs=read_legacy_catalogs(legacy); records=read_legacy_assessments(legacy,catalogs)
  plan=reconcile_legacy_assessments(legacy,catalogs,records)
  with AssessmentProjectFile.open(root/'test_projects/native'/native) as project:
   for record in (*records,*plan.recovered):
    result=project.load_saved_result(DocumentKey(project.metadata.project_id,record.ownership.document_id))
    cached=record.cumulative_damage if hasattr(record,'cumulative_damage') else record.summary['total_cumulative_damage']
    if cached is not None and round(result.damage,4)!=float(cached):
     kind=record.ownership.kind.value
     mismatches[kind]=mismatches.get(kind,0)+1
 report.append({'project':native,'assessment_versions_equal_to_original_calculators':totals,
                'task_rows_equal_to_original_calculators':tasks_checked,
                'old_cached_totals_disagree_after_four_place_rounding':mismatches})
(root/'artifacts/cleanup-m1r-original-oracle.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))

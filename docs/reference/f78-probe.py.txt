from pathlib import Path
import tempfile
from PyQt5.QtWidgets import QApplication
from PyQt5.QtTest import QTest
from PyQt5.QtCore import QPoint
from PyQt5.QtGui import QPainter, QPixmap
from ergotools.application.settings import Language
from ergotools.bootstrap import create_window
from ergotools.application.document_lifecycle import CloseChoice
from ergotools.domain.project import ProjectMetadata
from ergotools.domain.model_units import MeasurementSystem
from ergotools.domain.assessments.lifft_draft import LifftDraft,LifftTask
app=QApplication([]);out=Path.cwd()/'artifacts/vtk-body';out.mkdir(parents=True,exist_ok=True)
def capture(window,name):
 QTest.qWait(1000)
 renderer=window._body_renderer
 from vtkmodules.vtkRenderingCore import vtkWindowToImageFilter
 from vtkmodules.vtkIOImage import vtkPNGWriter
 pixmap=window.grab()
 renderer.render_window.Render()
 image_filter=vtkWindowToImageFilter();image_filter.SetInput(renderer.render_window);image_filter.ReadFrontBufferOff();image_filter.Update()
 writer=vtkPNGWriter();buffer=out/(name+'-vtk.png');writer.SetFileName(str(buffer));writer.SetInputConnection(image_filter.GetOutputPort());writer.Write()
 painter=QPainter(pixmap);painter.drawPixmap(renderer.widget.mapTo(window,QPoint(0,0)),QPixmap(str(buffer)));painter.end();pixmap.save(str(out/(name+'.png')))
with tempfile.TemporaryDirectory() as directory:
 window=create_window(Path(directory)/'settings.json',original_navigation=True,enable_vtk=True)
 window.show();capture(window,'initial')
 window.presenter.workspace.create(str(Path(directory)/'avatar.ergprj'),ProjectMetadata('p','Avatar',MeasurementSystem.IMPERIAL),'lifft',LifftDraft(MeasurementSystem.IMPERIAL,(LifftTask('a',12,20,100),)))
 window.refresh();window.run(window.calculate);capture(window,'calculated')
 renderer=window._body_renderer
 print('Actors',len(renderer.actors),'camera',renderer.renderer.GetActiveCamera().GetPosition(),'viewport',renderer.widget.width(),renderer.widget.height())
 window.body_panel.view_buttons[0].click();capture(window,'rotated')
 window.body_panel.view_buttons[1].click();capture(window,'zoomed')
 window.body_panel.view_buttons[2].click();capture(window,'reset-full-body')
 window.language=Language.SPANISH;window.translate();window.refresh();window.run(window.calculate);capture(window,'spanish')
 window.task_panel.reset_button.click();capture(window,'reset-inputs')
 window.presenter.workspace.close(choice=CloseChoice.DISCARD);window.close();QTest.qWait(100)
 print('VTK native scene/control/close probe complete; native framebuffer composited over Qt widget capture')

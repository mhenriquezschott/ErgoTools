import sys,tempfile
from dataclasses import replace
from pathlib import Path
from PyQt5.QtWidgets import QApplication,QMessageBox
from PyQt5.QtCore import Qt,QTimer
from PyQt5.QtTest import QTest
from ergotools.bootstrap import load_icons
from ergotools.domain.assessments.document import AssessmentKind
from ergotools.domain.assessments.lifft_draft import LifftDraft
from ergotools.domain.job_catalog import Job,JobCatalog,JobPlacement,WorkerJobAssignment
from ergotools.domain.job_risk import JobRiskMeasurement
from ergotools.domain.language import Language
from ergotools.domain.model_units import MeasurementSystem
from ergotools.domain.project import ProjectMetadata
from ergotools.domain.work_catalog import StationKey,WorkCatalog,WorkerEntry
from ergotools.domain.work_hierarchy import WorkHierarchy,SectionKey,LineKey
from ergotools.infrastructure.assessment_project_file import AssessmentProjectFile
from ergotools.presentation.desktop.rotation_dialog import RotationDialog
app=QApplication.instance() or QApplication([])
metadata=ProjectMetadata('p','Wheel JROT',MeasurementSystem.IMPERIAL)
station=StationKey('P','S','L','1')
hierarchy=WorkHierarchy(WorkCatalog('p',(WorkerEntry('w'),WorkerEntry('x')),(station,),('Day',)),('P',),(SectionKey('P','S'),),(LineKey('P','S','L'),))
owner=hierarchy.work.worker_context('w',station.at_shift('Day'))
jobs=JobCatalog('p',(Job('j','Job ñ'),Job('k','Other')),(JobPlacement(1,'j',owner.workplace),),(WorkerJobAssignment(1,owner,1),))

from time import monotonic
from ergotools.infrastructure.pulp_rotation_optimizer import PulpRotationOptimizer
from ergotools.domain.rotation_scheme import RotationMode

def wait(controller):
 deadline=monotonic()+10
 while controller.busy and monotonic()<deadline:QTest.qWait(5)
 assert not controller.busy
with tempfile.TemporaryDirectory() as directory:
 for language in Language:
  path=Path(directory)/(language.value+'.ergprj')
  with AssessmentProjectFile.create_with_document(path,metadata,'source',LifftDraft(MeasurementSystem.IMPERIAL),format_version=7) as project:
   project.save_hierarchy(hierarchy);project.save_jobs(jobs)
   for identity,job,damage,probability,unit in [(1,'j',0,0,'Imperial'),(2,'k',.2,80,'Metric')]:
    risk=project.create_job_risk_draft(project.load_job_risk(),job,identity)
    profile=replace(risk.profile(identity),measurements=(JobRiskMeasurement(AssessmentKind.LIFFT,damage,probability,unit),JobRiskMeasurement(AssessmentKind.DUET,.1 if identity==1 else .3,20 if identity==1 else 60,"Imperial"),JobRiskMeasurement(AssessmentKind.TST,.05 if identity==1 else .4,10 if identity==1 else 90,"Metric")))
    project.approve_job_risk(risk,identity,replacement=profile)
   win=RotationDialog(project,load_icons(),language=language,optimizer=PulpRotationOptimizer());win._message=lambda *args,**kwargs:None
   win.show();app.processEvents();win.workers_combo.setCurrentText('2');win.blocks_combo.setCurrentText('2')
   QTest.mouseClick(win.apply_scope_button,Qt.LeftButton)
   win.scheme_combo.setEditText('Wheel CBC ñ')
   for row,worker in enumerate(('w','x')):
    win.grid.set_choice(row,0,worker)
    for column in (1,2):win.grid.set_choice(row,column,win.grid.pool.targets[row].target.key)
   original=win.grid.current_scheme('Wheel CBC ñ');before=path.read_bytes()
   QTest.mouseClick(win.optimize_button,Qt.LeftButton);wait(win.optimization)
   assert win.optimization.preview is not None
   assert all(win.optimized_table.item(row,3).text()=='40.0%' for row in range(2))
   assert win.grid.current_scheme('Wheel CBC ñ')==original and path.read_bytes()==before
   assert project.load_rotation_schemes()==()
   QTest.mouseClick(win.compare_button,Qt.LeftButton)
   comparison=win.optimization.comparison
   assert comparison is not None and comparison.tabs is None
   assert comparison.tables[AssessmentKind.DUET][1].item(0,3).text()=='40.0%'
   comparison.close()
   QTest.mouseClick(win.optimize_all_button,Qt.LeftButton);wait(win.optimization)
   comparison=win.optimization.comparison
   assert comparison is not None and comparison.tabs.count()==3
   assert win.optimization.preview is None
   for index,kind in enumerate(AssessmentKind):
    comparison.tabs.setCurrentIndex(index)
    expected='50.0%' if kind is AssessmentKind.TST else '40.0%'
    assert comparison.tables[kind][1].item(0,3).text()==expected
   assert win.grid.current_scheme('Wheel CBC ñ')==original and path.read_bytes()==before
   QTest.mouseClick(comparison.apply_button,Qt.LeftButton)
   assert win.grid.current_scheme('Wheel CBC ñ').current==win.optimization.preview.current
   assert path.read_bytes()==before
   QTest.mouseClick(win.save_button,Qt.LeftButton)
   saved,=project.load_rotation_schemes()
   assert saved.mode is RotationMode.ALL_TOOLS and saved.primary_kind is None
   assert saved.current==win.optimization.preview.current
   win.close();win.deleteLater();app.processEvents()
  with AssessmentProjectFile.open(path) as reopened:
   assert reopened.load_rotation_schemes()==(saved,)
   assert reopened.load_job_risk().profile(1).measurements[0].unit=='Imperial'
   assert reopened.load_job_risk().profile(2).measurements[0].unit=='Metric'
for name,module in tuple(sys.modules.items()):
 if name=='ergotools' or name.startswith('ergotools.'):
  filename=getattr(module,'__file__',None)
  if filename:assert Path(filename).is_relative_to(Path(sys.argv[1])),filename
print('Extracted-wheel native EN/ES actual CBC/other-tool comparison and HiGHS/all tabs/Apply/explicit Save/reopen passed; exact links and mixed unit evidence retained; no implicit writes.')
